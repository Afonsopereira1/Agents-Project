 23class Source:
 24    """Perfil de perceção de um sensor: canal, alcance, taxas de erro e confiança.
 25
 26    Attributes:
 27        kind (str): Canal: ``acoustic``, ``visual`` ou ``human``.

 34        position_error (int): Ruído máximo da posição lida, em células por eixo.
 35        confidence (float): Confiança atribuída às observações geradas, em [0, 1].
 36    """
 37    kind: str
 38    detection_range: float
 39    sensitivity: float = 1.0
 40    false_positive_rate: float = 0.0
 41    false_negative_rate: float = 0.0
 42    position_error: int = 0
 43    confidence: float = 0.8

 45    def __post_init__(self):
 46        """Validar canal, alcance, taxas e confiança em [0, 1] e erro de posição.
 47
 48        Raises:
 49            ValueError: Se o canal for desconhecido, um valor for negativo ou não finito,

 66class Event:
 67    """Evento de referência (threat, thunder ou tourist), fixo ou em movimento.
 68
 69    Attributes:
 70        event_id (str): Identificador único do evento.
 71        kind (str): Tipo real: ``threat``, ``thunder`` ou ``tourist``.
 72        position (Position): Posição inicial.
 73        starts_at (float): Instante de início, em segundos simulados.
 74        duration (float): Duração, em segundos simulados; positiva.
 75        destination (Position | None): Destino do movimento; ``None`` se o evento é fixo.
 76        speed (float): Velocidade em direção ao destino, em células por segundo.
 77    """
 78    event_id: str
 79    kind: str
 80    position: Position
 81    starts_at: float
 82    duration: float
 83    destination: Position | None = None
 84    speed: float = 1.0
 85
 86    def __post_init__(self):
 87        """Validar identificador, tipo, posições e tempos; exige duração positiva.
 88
 89        Raises:
 90            ValueError: Se o identificador estiver vazio, o tipo for desconhecido, um tempo
 91                ou a velocidade for negativo ou não finito, ou o fim não for finito e posterior
 92                ao início.
 93            TypeError: Se ``position`` ou ``destination`` não forem ``Position``.
 94        """
 95        if not isinstance(self.event_id, str) or not self.event_id.strip():
 96            raise ValueError("O evento precisa de um identificador.")
 97        if not isinstance(self.kind, str) or self.kind not in CHANNELS:
 98            raise ValueError("Evento deve ser threat, thunder ou tourist.")
 99        if not isinstance(self.position, Position):
100            raise TypeError("A posição do evento deve ser Position.")
101        if self.destination is not None and not isinstance(self.destination, Position):
102            raise TypeError("O destino deve ser Position.")
103        for name in ("starts_at", "duration", "speed"):
104            nonnegative(getattr(self, name), name)
105        try:
106            end = self.starts_at + self.duration
107        except OverflowError as error:
108            raise ValueError("Duração do evento fora da capacidade do relógio.") from error
109        if not self.starts_at < end < float("inf"):
110            raise ValueError("O evento precisa de duração positiva e fim finito.")

112    def active(self, now: float) -> bool:
113        """Indicar se ``now`` pertence a [starts_at, starts_at + duration[.
114
115        Args:
116            now (float): Instante, em segundos simulados.
117
118        Returns:
119            bool: ``True`` se o evento estiver ativo nesse instante.
120        """
121        return self.starts_at <= now < self.starts_at + self.duration
122
123    def position_at(self, now: float) -> Position:
124        """Devolver a posição real do evento em ``now``.
125
126        Sem destino ou antes do início é a posição inicial; depois do fim mantém
127        a posição alcançada no último instante da duração.
128
129        Args:
130            now (float): Instante, em segundos simulados.
131
132        Returns:
133            Position: Célula onde o evento está.
134        """
135        elapsed = min(self.duration, max(0, now - self.starts_at))
136        return position_at(self.position, self.destination or self.position, self.speed, elapsed)

179class Scenario:
180    """O avaliador conserva este objeto; agentes recebem apenas source(id).
181
182    Attributes:
183        world (World): Mundo do cenário, com o relógio da execução.
184    """
185
186    def __init__(self, world: World, sources: dict[str, Source], events: tuple[Event, ...],
187                 seed: int, corroboration_window: float = 5.0, observation_prefix: str = "obs"):
188        """Inicializar o cenário: validar mundo, fontes e eventos e amostrar o instante zero.
189
190        Fontes e eventos ficam ordenados por identificador, pelo que a amostragem
191        não depende da ordem de entrada.
192
193        Args:
194            world (World): Mundo com o relógio ainda em zero.
195            sources (dict[str, Source]): Perfil de cada sensor do mundo, exatamente um por
196                sensor.
197            events (tuple[Event, ...]): Eventos de referência, com identificadores únicos.
198            seed (int): Semente do gerador aleatório da amostragem.
199            corroboration_window (float): Largura da janela dos feeds, em segundos simulados.
200            observation_prefix (str): Prefixo dos identificadores das observações geradas.
201
202        Raises:
203            TypeError: Se ``world`` não for ``World`` ou houver perfis ou eventos de tipo
204                errado.
205            ValueError: Se o relógio não estiver em zero, ``seed`` não for inteira, a janela
206                for inválida, as fontes não coincidirem com os sensores, houver eventos
207                repetidos ou posições fora do mapa.
208        """
209        if not isinstance(world, World):
210            raise TypeError("world deve ser World.")
211        if world.clock.now != 0:
212            raise ValueError("O cenário deve começar com o relógio em zero.")
213        if type(seed) is not int:
214            raise ValueError("seed deve ser um inteiro.")
215        nonnegative(corroboration_window, "corroboration_window")
216        if not isinstance(sources, dict) or sources.keys() != world.sensors.keys():
217            raise ValueError("Cada sensor deve ter exatamente um perfil de fonte.")
218        if any(not isinstance(source, Source) for source in sources.values()):
219            raise TypeError("Os perfis devem ser Source.")
220        events = tuple(events)
221        identifiers = set()
222        for event in events:
223            if not isinstance(event, Event):
224                raise TypeError("Os eventos devem ser Event.")
225            if event.event_id in identifiers:
226                raise ValueError("Identificador de evento repetido.")
227            identifiers.add(event.event_id)
228            world.validate_position(event.position)
229            if event.destination is not None:
230                world.validate_position(event.destination)
231        self.world = world
232        self._sources = dict(sorted(sources.items()))
233        self._events = tuple(sorted(events, key=lambda event: event.event_id))
234        self._rng = random.Random(seed)
235        # IDs únicos entre execuções permitem à reputação aprender sem colisões.
236        self._observation_prefix = observation_prefix
237        self._feeds = {key: ObservationFeed(world.clock, corroboration_window)
238                       for key in self._sources}
239        self._attributions: dict[str, str] = {}
240        self._steps: list[float] = []
241        self._sampled_at = 0.0
242        self._config = {
243            "width": world.width, "height": world.height,
244            "sensors": {key: model_to_dict(pos) for key, pos in world.sensors.items()},
245            "patrols": [model_to_dict(patrol) for patrol in world.patrols],
246            "scenario": {
247                "seed": seed, "corroboration_window": corroboration_window,
248                "sources": {key: model_to_dict(source) for key, source in self._sources.items()},
249                "events": [model_to_dict(event) for event in self._events],
250            },
251        }
252        self._sample()
253
254    @classmethod
255    def from_config(cls, path: str | Path) -> "Scenario":
256        """Ler mundo e cenário de um JSON; aqui a secção ``scenario`` é obrigatória.
257
258        Exige exatamente seed, corroboration_window, sources e events.
259
260        Args:
261            path (str | Path): Ficheiro JSON com o mundo e a secção ``scenario``.
262
263        Returns:
264            Scenario: Cenário pronto, já amostrado no instante zero.
265
266        Raises:
267            ValueError: Se houver chaves em falta, a mais ou repetidas, ou tipos errados.
268        """
269        with open(path, encoding="utf-8") as source:
270            data = json.load(source, object_pairs_hook=_unique_keys)
271        world = World.from_dict(data)
272        try:
273            config = data["scenario"]
274            if set(config) != {"seed", "corroboration_window", "sources", "events"}:
275                raise ValueError("scenario exige seed, corroboration_window, sources e events.")
276            if not isinstance(config["sources"], dict) or not isinstance(config["events"], list):
277                raise ValueError("sources deve ser um objeto e events uma lista.")
278            sources = {key: Source(**value) for key, value in config["sources"].items()}
279            events = tuple(event_from_dict(item) for item in config["events"])
280            return cls(world, sources, events, config["seed"], config["corroboration_window"])
281        except (KeyError, TypeError) as error:
282            raise ValueError("Configuração de cenário inválida ou incompleta.") from error

298    def advance(self, seconds: float = 1.0) -> float:
299        """Avançar e amostrar no novo instante; eventos saltados não geram leituras.
300
301        Args:
302            seconds (float): Duração do avanço, em segundos simulados.
303
304        Returns:
305            float: Novo instante atual.
306
307        Raises:
308            ValueError: Se o relógio tiver sido avançado fora do cenário ou ``seconds`` não
309                for finito e positivo.
310        """
311        if self.world.clock.now != self._sampled_at:
312            raise ValueError("Usar scenario.advance() para manter tempo e amostras sincronizados.")
313        now = self.world.clock.advance(seconds)
314        self._steps.append(seconds)
315        self._sample()
316        self._sampled_at = now
317        return now

319    def inspect(self, position: Position, now: float, task: str) -> str:
320        """Inspecionar localmente, de forma idealizada, uma posição.
321
322        Não expõe eventos nem ground_truth: só classifica a célula visitada no
323        instante indicado. A intervenção do guarda só é bem-sucedida se a ameaça
324        ainda lá estiver; um guarda que chega tarde devolve not_confirmed. Não
325        remove eventos globais.
326        Sensibilidade e falhas de inspeção ficam para os cenários posteriores.
327
328        Args:
329            position (Position): Célula inspecionada.
330            now (float): Instante da inspeção, em segundos simulados; não pode ser futuro.
331            task (str): Tarefa da missão: ``verify_incident`` ou ``respond_incident``.
332
333        Returns:
334            str: ``confirmed`` se houver uma ameaça ativa na célula, ``not_confirmed`` se
335                não, ou ``failed`` para uma tarefa desconhecida.
336
337        Raises:
338            ValueError: Se a posição estiver fora do mapa ou ``now`` for inválido ou futuro.
339        """
340        self.world.validate_position(position)
341        nonnegative(now, "now")
342        if now > self.world.clock.now:
343            raise ValueError("A inspeção não pode consultar o futuro.")
344        if task not in ("verify_incident", "respond_incident"):
345            return "failed"
346        return "confirmed" if any(
347            event.kind == "threat" and event.active(now) and event.position_at(now) == position
348            for event in self._events
349        ) else "not_confirmed"
350
351    def _sample(self) -> None:
352        """Gerar as observações do instante atual para cada evento ativo e cada fonte.
353
354        Só contam fontes de canal compatível com o evento dentro do alcance
355        euclidiano. A probabilidade de leitura é sensitivity × (1 −
356        false_negative_rate) para ameaças e sensitivity × false_positive_rate
357        para os outros eventos. A posição observada leva ruído inteiro até
358        ``position_error`` por eixo, limitado ao mapa. A ligação entre observação
359        e evento fica apenas em ``_attributions``, para o avaliador.
360        """
361        now = self.world.clock.now
362        for event in self._events:
363            if not event.active(now):
364                continue
365            real = event.position_at(now)
366            for source_id, source in self._sources.items():
367                origin = self.world.sensors[source_id]
368                if (source.kind not in CHANNELS[event.kind]
369                        or hypot(real.x - origin.x, real.y - origin.y) > source.detection_range):
370                    continue
371                probability = source.sensitivity * (
372                    1 - source.false_negative_rate if event.kind == "threat"
373                    else source.false_positive_rate
374                )
375                if self._rng.random() >= probability:
376                    continue
377                error = source.position_error
378                position = Position(
379                    min(self.world.width - 1, max(0, real.x + self._rng.randint(-error, error))),
380                    min(self.world.height - 1, max(0, real.y + self._rng.randint(-error, error))),
381                )
382                obs_id = f"{self._observation_prefix}-{len(self._attributions) + 1:06d}"
383                observation = Observation(obs_id, source_id, SIGNALS[source.kind], position,
384                                          now, source.confidence)
385                self._feeds[source_id]._history.append(observation)
386                self._attributions[obs_id] = event.event_id

389    def ground_truth(self) -> dict:
390        """Devolver uma cópia da verdade para avaliação.
391
392        Returns:
393            dict: ``events`` (rótulo, posição real e estado de cada evento) e

405        }

407    def save_run(self, path: str | Path) -> None:
408        """Gravar a configuração inicial, os avanços e os resultados.
409
410        Nunca substitui uma execução: o ficheiro é aberto em modo ``x``.
411
412        Args:
413            path (str | Path): Novo ficheiro JSON; não pode existir.
414
415        Raises:
416            ValueError: Se o relógio tiver sido avançado fora do cenário.
417        """
418        if self.world.clock.now != self._sampled_at:
419            raise ValueError("O relógio foi avançado fora do cenário.")
420        data = {"config": self._config, "steps": self._steps,
421                "finished_at": self.world.clock.now, "ground_truth": self.ground_truth,
422                "observations": [model_to_dict(obs) for feed in self._feeds.values()
423                                 for obs in feed._history]}
424        with open(path, "x", encoding="utf-8") as output:
425            json.dump(data, output, ensure_ascii=False, indent=2, allow_nan=False)
426            output.write("\n")

352    def _sample(self) -> None:
353        """Gerar as observações do instante atual para cada evento ativo e cada fonte.
354
355        Só contam fontes de canal compatível com o evento dentro do alcance
356        euclidiano. A probabilidade de leitura é sensitivity × (1 −
357        false_negative_rate) para ameaças e sensitivity × false_positive_rate
358        para os outros eventos. A posição observada leva ruído inteiro até
359        ``position_error`` por eixo, limitado ao mapa. A ligação entre observação
360        e evento fica apenas em ``_attributions``, para o avaliador.
361        """
362        now = self.world.clock.now
363        for event in self._events:
364            if not event.active(now):
365                continue
366            real = event.position_at(now)
367            for source_id, source in self._sources.items():
368                origin = self.world.sensors[source_id]
369                if (source.kind not in CHANNELS[event.kind]
370                        or hypot(real.x - origin.x, real.y - origin.y) > source.detection_range):
371                    continue
372                probability = source.sensitivity * (
373                    1 - source.false_negative_rate if event.kind == "threat"
374                    else source.false_positive_rate
375                )
376                if self._rng.random() >= probability:
377                    continue
378                error = source.position_error
379                position = Position(
380                    min(self.world.width - 1, max(0, real.x + self._rng.randint(-error, error))),
381                    min(self.world.height - 1, max(0, real.y + self._rng.randint(-error, error))),
382                )
383                obs_id = f"{self._observation_prefix}-{len(self._attributions) + 1:06d}"
384                observation = Observation(obs_id, source_id, SIGNALS[source.kind], position,
385                                          now, source.confidence)
386                self._feeds[source_id]._history.append(observation)
387                self._attributions[obs_id] = event.event_id
