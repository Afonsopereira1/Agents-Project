 41def fault_settings(faults):
 42    """Validar a injeção de falhas do transporte local (Etapa 8).
 43
 44    Args:
 45        faults (dict | None): Chaves de ``FAULT_DEFAULTS`` a alterar (``seed``, ``loss``,
 46            ``delay``, ``drop``, ``duplicate``, ``unavailable_agents``,
 47            ``occupied_after_proposal``, ``flood``); None desliga as falhas.
 48
 49    Returns:
 50        dict | None: Cópia independente com as omissões preenchidas, ou None sem falhas.
 51
 52    Raises:
 53        ValueError: Se houver chaves desconhecidas, valores fora do intervalo ou nomes de
 54            mensagens, agentes ou recursos que não existem no MVP.
 55    """
 56    if faults is None:
 57        return None
 58    if not isinstance(faults, dict) or set(faults) - set(FAULT_DEFAULTS):
 59        raise ValueError(f"faults aceita apenas {sorted(FAULT_DEFAULTS)}.")
 60    settings = {**FAULT_DEFAULTS, **faults}
 61    if type(settings["seed"]) is not int:
 62        raise ValueError("faults.seed deve ser inteiro.")
 63    if type(settings["loss"]) not in (int, float) or not 0 <= settings["loss"] <= 1:
 64        raise ValueError("faults.loss deve estar entre 0 e 1.")
 65    if not isinstance(settings["delay"], dict) or any(
 66            type(value) not in (int, float) or not 0 <= value < 60 for value in settings["delay"].values()):
 67        raise ValueError("faults.delay associa tipos de mensagem a segundos em [0, 60[.")
 68    if type(settings["flood"]) is not int or settings["flood"] < 0:
 69        raise ValueError("faults.flood deve ser um inteiro não negativo.")
 70    for key in ("drop", "duplicate", "unavailable_agents", "occupied_after_proposal"):
 71        if not isinstance(settings[key], list) or any(not isinstance(v, str) for v in settings[key]):
 72            raise ValueError(f"faults.{key} deve ser uma lista de nomes.")
 73    message_types = {kind.value for kind in MessageType}
 74    for key, supported in (
 75        ("delay", message_types), ("drop", message_types), ("duplicate", message_types),
 76        ("unavailable_agents", MVP_AGENT_IDS), ("occupied_after_proposal", MVP_RESOURCE_IDS),
 77    ):
 78        if any(value not in supported for value in settings[key]):
 79            raise ValueError(f"faults.{key} aceita apenas {', '.join(sorted(supported))}.")
 80    # Os parâmetros fornecidos e os defaults têm o mesmo isolamento por execução.
 81    return deepcopy(settings)

 92def validate_experimental_settings(*, xmpp=False, corroboration_settings=None,
 93                                   receive_timeout=None, minimum_autonomy_percent=10.0,
 94                                   selection_weights=None):
 95    """Validar o Bloco G antes de criar agentes; devolver os valores efetivos.
 96
 97    A janela de corroboração usa tempo simulado. ``receive_timeout`` é o prazo
 98    base em tempo real e não o teto dinâmico opcional do Bloco F.
 99
100    Args:
101        xmpp (bool): Transporte XMPP; muda o prazo por omissão de 0.5 s para 2.0 s.
102        corroboration_settings (dict[str, float] | None): ``max_distance`` (células) e
103            ``max_time_difference`` (s simulados); None usa ``CORROBORATION_DEFAULTS``.
104        receive_timeout (float | None): Prazo base, em segundos reais; None usa a omissão
105            do transporte.
106        minimum_autonomy_percent (float): Reserva mínima de energia, em % de 0 a 100.
107        selection_weights (dict[str, float] | None): Pesos e escalas de seleção de
108            propostas; None mantém as omissões do iniciador.
109
110    Returns:
111        tuple[dict[str, float], float, float]: Parâmetros de corroboração, prazo base e
112            reserva mínima efetivos.
113
114    Raises:
115        ValueError: Se houver chaves desconhecidas, valores negativos ou não finitos, prazo
116            não positivo, reserva fora de [0, 100], escala nula ou todos os pesos a zero.
117    """
118    if corroboration_settings is not None and (
119            not isinstance(corroboration_settings, dict)
120            or set(corroboration_settings) - set(CORROBORATION_DEFAULTS)):
121        raise ValueError(f"corroboration_settings aceita apenas {sorted(CORROBORATION_DEFAULTS)}.")
122    corroboration = {**CORROBORATION_DEFAULTS, **(corroboration_settings or {})}
123    for name, value in corroboration.items():
124        if not is_finite_number(value) or value < 0:
125            raise ValueError(f"corroboration_settings.{name} deve ser finito e não negativo.")
126    timeout = (2.0 if xmpp else 0.5) if receive_timeout is None else receive_timeout
127    if not is_finite_number(timeout) or timeout <= 0:
128        raise ValueError("receive_timeout deve ser finito e positivo.")
129    if not is_finite_number(minimum_autonomy_percent) or not 0 <= minimum_autonomy_percent <= 100:
130        raise ValueError("minimum_autonomy_percent deve estar entre 0 e 100.")
131    if selection_weights is not None:
132        defaults = {"eta_weight": 1.0, "energy_weight": 1.0, "load_weight": 1.0,
133                    "eta_scale_seconds": 60.0, "energy_scale_percent": 100.0, "load_scale": 1.0}
134        if not isinstance(selection_weights, dict) or set(selection_weights) - set(defaults):
135            raise ValueError(f"selection_weights aceita apenas {sorted(defaults)}.")
136        cost = {**defaults, **selection_weights}
137        for name, value in cost.items():
138            if not is_finite_number(value) or value < 0 or ("scale" in name and value == 0):
139                raise ValueError(f"selection_weights.{name} inválido: peso não negativo, escala positiva.")
140        if not any(cost[name] > 0 for name in ("eta_weight", "energy_weight", "load_weight")):
141            raise ValueError("Pelo menos um peso deve ser positivo.")
142    return corroboration, timeout, minimum_autonomy_percent

145def scenario_settings(scenario):
146    """Montar fontes, eventos e recursos de um cenário da Etapa 9.
147
148    O mapa não é criado aqui: vem de ``map_size`` em ``run_mvp`` (8x8 por omissão).
149
150    Args:
151        scenario (dict | None): Cenário com ``run_id``, ``seed``, ``sources``, ``events`` e
152            ``resources``; None usa o cenário base do MVP (uma ameaça fixa em (3, 2)).
153
154    Returns:
155        tuple: Semente, fontes (``dict[str, Source]``), eventos (``tuple[Event, ...]``),
156            opções por recurso (``dict``) e prefixo dos identificadores das observações.
157
158    Raises:
159        ValueError: Se faltarem ou sobrarem chaves, o ``run_id`` estiver vazio, a semente
160            não for inteira, fontes, eventos ou recursos forem inválidos, ou algum evento
161            não começar em t=0.
162    """
163    if scenario is None:
164        return (SCENARIO_SEED, {key: Source("acoustic", 5) for key in ("sensor-1", "sensor-2")},
165                (Event("event-mvp", "threat", Position(3, 2), 0, 1000),), {}, "obs")
166    if not isinstance(scenario, dict) or set(scenario) != {"run_id", "seed", "sources", "events", "resources"}:
167        raise ValueError("scenario exige run_id, seed, sources, events e resources.")
168    if not isinstance(scenario["run_id"], str) or not scenario["run_id"].strip():
169        raise ValueError("scenario.run_id deve identificar a execução.")
170    if type(scenario["seed"]) is not int:
171        raise ValueError("scenario.seed deve ser inteiro.")
172    resources = scenario["resources"]
173    if not isinstance(resources, dict) or set(resources) - set(MVP_RESOURCES) or any(
174            not isinstance(v, dict) or set(v) - RESOURCE_KEYS for v in resources.values()):
175        raise ValueError(f"scenario.resources aceita {sorted(MVP_RESOURCES)} com {sorted(RESOURCE_KEYS)}.")
176    try:
177        sources = {key: Source(**value) for key, value in scenario["sources"].items()}
178        events = tuple(event_from_dict(item) for item in scenario["events"])
179    except (AttributeError, KeyError, TypeError) as error:
180        raise ValueError("Fontes ou eventos do cenário inválidos.") from error
181    if any(event.starts_at != 0 for event in events):
182        # O sensor só lê o feed em t=0 (janela única); um evento posterior seria um FN falso.
183        raise ValueError("Nesta janela única os eventos têm de começar em t=0.")
184    return scenario["seed"], sources, events, resources, f"{scenario['run_id']}-obs"

187async def run_mvp(*, xmpp=False, password="local-demo", inspection_outcomes=None,
188                  interrupt_resource=None, max_seconds=None,
189                  response_settings=None, reputation=None, faults=None,
190                  scenario=None, corroborate=True, selection_weights=None, run_id=None,
191                  extra_initiators=None, extra_patrols=None, cfp_limit=None, max_receive_timeout=2.0,
192                  queue_rounds=60, silent_invalid=True, queue_priority=False,
193                  corroboration_settings=None, receive_timeout=None, minimum_autonomy_percent=10.0,
194                  extra_rangers=None, extra_informants=None, map_size=(8, 8), sensor_positions=None,
195                  expire_unseen=False, queue_until_free=False, chain_missions=False, retransmit=0,
196                  alert_distance=None):
197    """Correr uma execução do MVP e devolver o relatório.
198
199    O runner só fornece ambiente, relógio e transporte; os agentes decidem. O transporte
200    local executa os mesmos comportamentos e contratos SPADE, mas não demonstra entrega
201    pela rede XMPP. ``max_receive_timeout``, ``queue_rounds`` e ``silent_invalid`` estão
202    ligados por omissão desde 18-11-2026; ``PUBLISHED_CONDITIONS`` desliga-os, como nos
203    resultados recolhidos.
204
205    Args:
206        xmpp (bool): Ligar os agentes a um servidor XMPP em localhost em vez do transporte
207            local.
208        password (str): Palavra-passe das contas dos agentes.
209        inspection_outcomes (dict[str, str] | None): Resultado fixo da inspeção por
210            ``resource_id``, para testes determinísticos; os outros recursos usam
211            ``Scenario.inspect``.
212        interrupt_resource (str | None): ``resource_id`` cuja missão é interrompida uma vez,
213            quando o recurso estiver fora da base; para testes determinísticos.
214        max_seconds (float | None): Limite global, em segundos reais. None calcula um que
215            cobre a verificação e as tentativas das duas fases de missão; um limite
216            explícito mantém o watchdog do chamador.
217        response_settings (dict | None): Ativa a Etapa 7 (``reputation``, ``policy``,
218            ``sources``, ``severity`` e, opcionais, ``severity_by_source``,
219            ``compatibility_rules`` e ``adjudication``). Com ``adjudication``, o avaliador
220            atualiza ``reputation`` após a execução a partir dos MISSION_RESULT com
221            cobertura suficiente.
222        reputation (ReputationBook | None): Apenas aprendizagem previamente validada pelo
223            avaliador; o sensor recebe uma cópia dos pesos. None cria uma nova a partir de
224            ``response_settings["reputation"]``.
225        faults (dict | None): Perda, atraso, duplicação e indisponibilidade só no transporte
226            local (ver ``fault_settings``); não altera nem valida o servidor XMPP.
227        scenario (dict | None): Cenário da Etapa 9, que substitui semente, fontes, eventos e
228            recursos do MVP (ver ``scenario_settings``). A sua identidade (``run_id``) é
229            também o âmbito da adjudicação.
230        corroborate (bool): False ignora a resposta do par.
231        selection_weights (dict[str, float] | None): Pesos passados a
232            ``select_best_proposal``, ex. só ETA para "mais próximo".
233        run_id (str | None): Identidade da execução; tem de coincidir com a do cenário quando
234            ambos são fornecidos. None usa a do cenário ou ``mvp-seed-<seed>``. Sem cenário,
235            repetir a mesma semente é replay para aprendizagem; uma nova amostra exige
236            ``run_id`` explícito, que não prova independência estatística.
237        extra_initiators (dict[str, Position] | None): Bloco F2: associa fontes do cenário a
238            posições; cada uma ganha um sensor no mapa e um IncidentSensorAgent próprio, que
239            só trata o que a sua fonte deteta. O runner não escolhe quem trata cada ameaça;
240            os campos de topo do relatório continuam a ser os do sensor-mvp.
241        extra_patrols (dict[str, Position] | None): C4 mínimo: associa novos IDs de patrulha
242            a posições base; são drones com 100 % de autonomia.
243        cfp_limit (int | None): K de C1: cada iniciador envia o CFP só aos K recursos com
244            base mais próxima do alvo (K × ronda nas rondas seguintes). None envia a todos.
245        max_receive_timeout (float | None): Teto do prazo dinâmico do iniciador, em segundos
246            reais (ver ``configure_mvp``). Um ``receive_timeout`` acima do teto exige um
247            teto maior (ou None).
248        queue_rounds (int): Rondas da fila do iniciador quando os recursos estão ocupados.
249        silent_invalid (bool): Os recursos ignoram um ACCEPT incoerente em vez de
250            responderem ``reservation_failed``.
251        queue_priority (bool): Ordenar a fila em cada recurso: primeiro a ocorrência mais
252            grave e, em caso de empate, a que espera há mais tempo. A gravidade de cada
253            iniciador vem de ``response_settings["severity"]`` ou, por fonte, de
254            ``response_settings["severity_by_source"]``.
255        corroboration_settings (dict[str, float] | None): ``max_distance`` e
256            ``max_time_difference``, com omissões 3.0 células e 2.0 s simulados.
257        receive_timeout (float | None): Prazo base, em segundos reais; None usa 0.5 s local
258            ou 2.0 s XMPP. Não liga o prazo dinâmico do Bloco F.
259        minimum_autonomy_percent (float): Reserva mínima de energia, em %, de todas as
260            patrulhas e do guarda.
261        extra_rangers (dict[str, Position] | None): Associa novos IDs de guarda a posições
262            base.
263        extra_informants (dict[str, Position] | None): Associa fontes do cenário a posições:
264            cada uma ganha um InformantAgent, e cada iniciador adicional passa a corroborar
265            com o informador mais próximo (sensor-2 incluído). Um dicionário vazio liga a
266            corroboração só com o sensor-2; None: os iniciadores adicionais não corroboram.
267        map_size (tuple[int, int]): Largura e altura do mapa (C4 completo), em células.
268        sensor_positions (dict[str, Position] | None): Posições que substituem as de
269            sensor-1 e/ou sensor-2 (C4 completo).
270        expire_unseen (bool): Prazo de validade do iniciador (ver ``configure_mvp``);
271            mitigação da espera na fila, desligada por omissão.
272        queue_until_free (bool): A fila espera até ao ``busy_until`` da recusa (ver
273            ``configure_mvp``); mitigação desligada por omissão.
274        chain_missions (bool): O recurso aceita a missão seguinte durante o regresso à base;
275            mitigação desligada por omissão.
276        retransmit (int): N: o iniciador reenvia até N vezes o mesmo VERIFY_REQUEST, CFP ou
277            ACCEPT quando a resposta não chega, e os recursos repetem a resposta já dada.
278            0 desliga.
279        alert_distance (float | None): D: liga o aviso entre iniciadores com cobertura
280            sobreposta. Cada um envia um ALERT aos vizinhos e, se dois leram a mesma ameaça
281            (até D células), só o que leu primeiro a trata; o outro fecha como
282            ``duplicate_alert``. O runner só diz quem é vizinho. None desliga.
283
284    Returns:
285        dict: Relatório da execução: configuração efetiva, ``ground_truth``, decisão, estado
286            e motivo do incidente, resultados, missões, recursos, falhas aplicadas,
287            atualizações de reputação, ``timeline`` e mensagens trocadas.
288
289    Raises:
290        ValueError: Se as definições forem inválidas, se ``faults``, iniciadores,
291            informadores ou recursos adicionais forem pedidos com XMPP, se ``run_id`` for
292            vazio ou diferente do do cenário, ou se houver IDs repetidos ou não suportados.
293        TimeoutError: Se a execução não terminar em ``max_seconds``.
294        Exception: A exceção de uma tarefa (comportamento ou relógio) que falhou, relançada.
295    """
296    corroboration, receive_timeout, minimum_autonomy_percent = validate_experimental_settings(
297        xmpp=xmpp, corroboration_settings=corroboration_settings, receive_timeout=receive_timeout,
298        minimum_autonomy_percent=minimum_autonomy_percent, selection_weights=selection_weights)
299    faults = fault_settings(faults)
300    if faults is not None and xmpp:
301        raise ValueError("A injeção de falhas só existe no transporte local.")
302    seed, sources, events, resource_settings, prefix = scenario_settings(scenario)
303    scenario_run_id = scenario["run_id"] if scenario is not None else None
304    if run_id is None:
305        run_id = scenario_run_id if scenario_run_id is not None else f"mvp-seed-{seed}"
306    if not isinstance(run_id, str) or not run_id.strip():
307        raise ValueError("run_id deve identificar a execução de avaliação.")
308    if scenario_run_id is not None and run_id != scenario_run_id:
309        raise ValueError("run_id deve coincidir com scenario.run_id.")
310    extra_initiators = dict(extra_initiators or {})
311    extra_patrols, extra_rangers = dict(extra_patrols or {}), dict(extra_rangers or {})
312    informants = dict(extra_informants or {})
313    if (extra_initiators or extra_patrols or extra_rangers or informants) and xmpp:
314        raise ValueError("Iniciadores, informadores e recursos adicionais só existem no transporte local.")
315    if (set(extra_patrols) | set(extra_rangers)) & set(MVP_RESOURCES) or set(extra_patrols) & set(extra_rangers):
316        raise ValueError("extra_patrols e extra_rangers exigem IDs diferentes entre si e de drone, vehicle e ranger.")
317    if ((set(extra_initiators) | set(informants)) & {"sensor-1", "sensor-2"}
318            or (set(extra_initiators) | set(informants)) - set(sources) or set(extra_initiators) & set(informants)):
319        raise ValueError("extra_initiators e extra_informants exigem fontes do cenário distintas, "
320                         "diferentes de sensor-1 e sensor-2.")
321    base_sensors = {"sensor-1": Position(2, 2), "sensor-2": Position(4, 2)}
322    if set(sensor_positions or {}) - set(base_sensors):
323        raise ValueError("sensor_positions aceita apenas sensor-1 e sensor-2.")
324    world = World(*map_size, {**base_sensors, **(sensor_positions or {}), **extra_initiators, **informants})
325    scenario = Scenario(world, sources, events, seed=seed, observation_prefix=prefix)
326    sensor = IncidentSensorAgent("sensor-mvp@localhost", password, "sensor-peer-mvp@localhost")
327    peer = InformantAgent("sensor-peer-mvp@localhost", password)
328    sensor.set_local_observation_feed(scenario.source("sensor-1"), 0.0)
329    sensor.set_corroboration_context(("sensor-2",) if corroborate else (),
330                                     corroboration["max_distance"], corroboration["max_time_difference"], 1,
331                                     compatibility_rules=(response_settings or {}).get("compatibility_rules"))
332    peers = {"sensor-2": peer}
333    for source_id in informants:
334        peers[source_id] = InformantAgent(f"{source_id}-mvp@localhost", password)
335        peers[source_id].set_local_observation_feed(scenario.source(source_id), source_id, 0.0)
336    extras = {}
337    for source_id, position in extra_initiators.items():
338        # C6: cada iniciador adicional pergunta ao informador mais próximo. Sem extra_informants
339        # não corrobora (o sensor-2 responde, mas a resposta não conta), como no Bloco F2.
340        # shortcut: os iniciadores adicionais não atualizam a reputação; alargar a adjudicação se for preciso.
341        nearest = min(peers, key=lambda key: distance(world.sensors[key], position))
342        extras[source_id] = extra = IncidentSensorAgent(
343            f"{source_id}-mvp@localhost", password, str(peers[nearest].jid))
344        extra.set_local_observation_feed(scenario.source(source_id), 0.0)
345        extra.set_corroboration_context(
346            (nearest,) if corroborate and extra_informants is not None else (),
347            corroboration["max_distance"], corroboration["max_time_difference"], 1,
348            compatibility_rules=(response_settings or {}).get("compatibility_rules"))
349    initiators = [sensor, *extras.values()]
350    if response_settings is not None:
351        if reputation is None:
352            reputation = ReputationBook(**response_settings["reputation"])
353        for source_id, initiator in zip(("sensor-1", *extras), initiators):
354            initiator.set_response_policy(
355                ResponsePolicy(**response_settings["policy"]),
356                reputation.snapshot(world.sensors), response_settings["sources"],
357                severity=response_settings.get("severity_by_source", {}).get(
358                    source_id, response_settings["severity"]),
359                default_reputation=reputation.score("sensor-1"),
360            )
361    peer.set_local_observation_feed(scenario.source("sensor-2"), "sensor-2", 0.0)
362    mobiles = []
363    for agent_class, name, capability in (
364        (DroneAgent, "drone", "mobile_verification"),
365        (VehicleAgent, "vehicle", "mobile_verification"),
366        (RangerAgent, "ranger", "incident_response"),
367        *((DroneAgent, name, "mobile_verification") for name in extra_patrols),
368        *((RangerAgent, name, "incident_response") for name in extra_rangers),
369    ):
370        options = resource_settings.get(name, {})
371        origin = (model_from_dict(options["position"], Position) if "position" in options
372                  else {**MVP_RESOURCES, **extra_patrols, **extra_rangers}[name])
373        world.validate_position(origin)
374        kind = "drone" if name in extra_patrols else "ranger" if name in extra_rangers else name
375        agent = agent_class(f"{name}-mvp@localhost", password,
376                            ResourceState(name, kind, origin, True,
