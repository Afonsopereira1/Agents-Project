isia.reasoning.corroboration

Critérios locais e configuráveis para corroborar observações.

Por onde começar:

  • select_eligible_sources: a quem pedir verificação.
  • aggregate_verification_responses: juntar e classificar as respostas.
  • evaluate_reputation_corroboration: score de suspeita ponderado pela reputação.
  • decide_preliminary_verification: observar ou pedir patrulha.

A verificação de compatibilidade usa só as duas observações e os limites configurados; nunca consulta a verdade do simulador nem os eventos reais.

  24def capability_supports_signal(capability: object, event_type: object) -> bool:
  25    """Reconhecer apenas correspondências modalidade-sinal definidas no simulador.
  26
  27    Esta regra verifica a adequação *do tipo* de fonte, não prova que cada
  28    deteção individual tenha sensibilidade perfeita nem consulta ground_truth.
  29    Sinais desconhecidos não justificam ausência informativa por omissão.
  30
  31    Args:
  32        capability (object): Modalidade declarada pela fonte (ex.: ``acoustic``).
  33        event_type (object): Tipo de sinal da hipótese.
  34
  35    Returns:
  36        bool: ``True`` se o simulador associar essa modalidade a esse sinal.
  37    """
  38    return (
  39        isinstance(capability, str)
  40        and isinstance(event_type, str)
  41        and SIGNALS.get(capability) == event_type
  42    )

  45def are_observations_compatible(
  46    first: Observation,
  47    second: Observation,
  48    max_distance: float,
  49    max_time_difference: float,
  50) -> bool:
  51    """Dizer se duas observações podem referir-se à mesma ocorrência.
  52
  53    Args:
  54        first (Observation): Primeira observação.
  55        second (Observation): Segunda observação.
  56        max_distance (float): Distância euclidiana máxima entre as posições (inclusiva).
  57        max_time_difference (float): Diferença máxima entre os instantes de observação
  58            (inclusiva), na mesma unidade de ``Observation.observed_at``.
  59
  60    Returns:
  61        bool: ``True`` se os ``event_id`` forem diferentes e os dois limites forem
  62            cumpridos; senão ``False``.
  63
  64    Raises:
  65        TypeError: Se uma observação ou um limite tiver um tipo inválido.
  66        ValueError: Se um limite for negativo ou não finito.
  67    """
  68    _validate_observation(first, "first")
  69    _validate_observation(second, "second")
  70    _validate_limit(max_distance, "max_distance")
  71    _validate_limit(max_time_difference, "max_time_difference")
  72
  73    if first.event_id == second.event_id:
  74        return False
  75
  76    time_difference = abs(first.observed_at - second.observed_at)
  77
  78    return positions_within_distance(first.position, second.position, max_distance) and (
  79        time_difference <= max_time_difference
  80    )

  83def positions_within_distance(first: Position, second: Position, max_distance: float) -> bool:
  84    """Comparar posições sem deixar escapar overflow na conversão de inteiros.
  85
  86    Args:
  87        first (Position): Primeira posição.
  88        second (Position): Segunda posição.
  89        max_distance (float): Distância euclidiana máxima (inclusiva).
  90
  91    Returns:
  92        bool: ``True`` se a distância entre as posições não exceder ``max_distance``.
  93
  94    Raises:
  95        TypeError: Se ``max_distance`` não for numérico.
  96        ValueError: Se ``max_distance`` for negativo ou não finito, ou se a distância
  97            exceder o intervalo numérico suportado.
  98    """
  99    _validate_limit(max_distance, "max_distance")
 100    try:
 101        distance = hypot(first.x - second.x, first.y - second.y)
 102    except OverflowError as error:
 103        raise ValueError("Position distance exceeds the supported numeric range.") from error
 104    return distance <= max_distance

 107def select_eligible_sources(
 108    observation: Observation,
 109    sensor_positions: Mapping[str, Position],
 110    source_profiles: Mapping[str, Source],
 111) -> tuple[str, ...]:
 112    """Devolver os sensores que podem ajudar a verificar a observação de forma independente.
 113
 114    Uma fonte é elegível se:
 115
 116    - tem posição e perfil ``Source`` conhecidos;
 117    - não é a fonte de ``observation``;
 118    - a observação está dentro do seu alcance de deteção (inclusive).
 119
 120    A modalidade da fonte não é comparada com o tipo da observação, de
 121    propósito: fontes acústicas, visuais e humanas complementam-se.
 122
 123    Usa só os metadados dos sensores; não consulta eventos do simulador, a
 124    verdade nem o histórico de outras fontes.
 125
 126    Args:
 127        observation (Observation): Observação a verificar.
 128        sensor_positions (Mapping[str, Position]): Posição de cada sensor, por ID.
 129        source_profiles (Mapping[str, Source]): Perfil de cada fonte, com o alcance de
 130            deteção.
 131
 132    Returns:
 133        tuple[str, ...]: IDs das fontes elegíveis, cada um uma vez, por ordem alfabética.
 134
 135    Raises:
 136        TypeError: Se ``observation`` não for ``Observation``, se um dos mapas não for
 137            ``Mapping``, ou se uma posição ou perfil tiver o tipo errado.
 138    """
 139    if not isinstance(observation, Observation):
 140        raise TypeError("observation must be an Observation.")
 141    if not isinstance(sensor_positions, Mapping):
 142        raise TypeError("sensor_positions must be a mapping.")
 143    if not isinstance(source_profiles, Mapping):
 144        raise TypeError("source_profiles must be a mapping.")
 145
 146    eligible = []
 147    for sensor_id in sorted(set(sensor_positions) & set(source_profiles)):
 148        if sensor_id == observation.source_id:
 149            continue
 150
 151        position = sensor_positions[sensor_id]
 152        profile = source_profiles[sensor_id]
 153        if not isinstance(position, Position):
 154            raise TypeError("Each sensor position must be a Position.")
 155        if not isinstance(profile, Source):
 156            raise TypeError("Each source profile must be a Source.")
 157
 158        if positions_within_distance(observation.position, position, profile.detection_range):
 159            eligible.append(sensor_id)
 160
 161    return tuple(eligible)

 164def classify_verification_response(
 165    initial_observation: Observation,
 166    response: Message,
 167    max_distance: float,
 168    max_time_difference: float,
 169    *,
 170    incompatibility_rules: Mapping[str, object] | None = None,
 171    compatibility_rules: Mapping[str, object] | None = None,
 172) -> dict[str, object]:
 173    """Classificar a evidência de um ``VERIFY_RESPONSE`` já validado.
 174
 175    Cada classificação exige:
 176
 177    - **compatível:** uma observação com ``event_id`` e ``source_id``
 178      diferentes dos da inicial e que cumpra :func:`are_observations_compatible`;
 179    - **contradição:** a resposta declara ``evidence_relation == "contradictory"``,
 180      a hipótese devolvida corresponde à inicial e há uma regra de
 181      incompatibilidade dada pelo chamador; se a declaração vier com hipótese
 182      não correlacionada, nenhuma observação conta como compatível;
 183    - **tipos de evento diferentes:** uma regra de compatibilidade dada pelo chamador;
 184    - **ausência informativa:** uma declaração explícita e validada de não
 185      deteção e cobertura. Uma resposta vazia ou um timeout nunca bastam.
 186
 187    Cada ``event_id`` é avaliado uma só vez. Não consulta o estado do
 188    simulador nem a verdade.
 189
 190    Args:
 191        initial_observation (Observation): Observação que originou o pedido.
 192        response (Message): ``VERIFY_RESPONSE`` a classificar.
 193        max_distance (float): Distância máxima para observações compatíveis (inclusiva).
 194        max_time_difference (float): Diferença temporal máxima (inclusiva).
 195        incompatibility_rules (Mapping[str, object] | None): Tipos de evento que
 196            contradizem cada tipo; ``None`` desativa contradições.
 197        compatibility_rules (Mapping[str, object] | None): Tipos de evento, além do
 198            próprio, compatíveis com cada tipo.
 199
 200    Returns:
 201        dict[str, object]: ``classification`` (``compatible``, ``contradictory``,
 202            ``informative_absence``, ``inconclusive`` ou ``no_data``) e os
 203            ``observation_ids`` usados para a obter, sem repetições; em evidência mista,
 204            também ``compatible_observation_ids`` e ``contradictory_observation_ids``.
 205
 206    Raises:
 207        TypeError: Se ``initial_observation`` ou ``response`` tiverem o tipo errado, ou
 208            se um limite ou as regras forem de tipo inválido.
 209        ValueError: Se um limite for negativo ou não finito, se a resposta for inválida
 210            ou se as regras contiverem strings vazias.
 211    """
 212    _validate_observation(initial_observation, "initial_observation")
 213    _validate_limit(max_distance, "max_distance")
 214    _validate_limit(max_time_difference, "max_time_difference")
 215    observations, status = _observations_from_response(response)
 216
 217    if status == "no_data":
 218        return {"classification": "no_data", "observation_ids": ()}
 219
 220    if status == "informative_absence":
 221        if _informative_absence_is_valid(initial_observation, response):
 222            return {
 223                "classification": "informative_absence",
 224                "observation_ids": (),
 225            }
 226        return {"classification": "inconclusive", "observation_ids": ()}
 227
 228    unique_observations = []
 229    seen_event_ids = set()
 230    for observation in observations:
 231        if observation.event_id in seen_event_ids:
 232            continue
 233        seen_event_ids.add(observation.event_id)
 234        if (
 235            observation.event_id != initial_observation.event_id
 236            and observation.source_id != initial_observation.source_id
 237        ):
 238            unique_observations.append(observation)
 239
 240    incompatible_types = _configured_types(
 241        initial_observation.event_type, incompatibility_rules, "incompatibility"
 242    )
 243    compatible_types = {initial_observation.event_type}
 244    compatible_types.update(
 245        _configured_types(initial_observation.event_type, compatibility_rules, "compatibility")
 246    )
 247    # A declaração de contradição refere-se à resposta; cada observação
 248    # continua a ser classificada pelo seu próprio tipo e pelas regras locais.
 249    contradictory_declaration = (
 250        response.payload.get("evidence_relation") == "contradictory"
 251    )
 252    explicit_contradiction = contradictory_declaration and _hypothesis_is_correlated(
 253        initial_observation, response
 254    )
 255    contradictory = [
 256        observation
 257        for observation in unique_observations
 258        if explicit_contradiction
 259        and observation.event_type in incompatible_types
 260        and are_observations_compatible(
 261            initial_observation,
 262            observation,
 263            max_distance,
 264            max_time_difference,
 265        )
 266    ]
 267    contradictory_ids = {observation.event_id for observation in contradictory}
 268    compatible = [
 269        observation
 270        for observation in unique_observations
 271        if observation.event_id not in contradictory_ids
 272        and (not contradictory_declaration or explicit_contradiction)
 273        and observation.event_type in compatible_types
 274        and are_observations_compatible(
 275            initial_observation,
 276            observation,
 277            max_distance,
 278            max_time_difference,
 279        )
 280    ]
 281    if compatible and contradictory:
 282        return {
 283            "classification": "inconclusive",
 284            "observation_ids": tuple(
 285                observation.event_id
 286                for observation in (*compatible, *contradictory)
 287            ),
 288            "compatible_observation_ids": tuple(
 289                observation.event_id for observation in compatible
 290            ),
 291            "contradictory_observation_ids": tuple(
 292                observation.event_id for observation in contradictory
 293            ),
 294        }
 295    if contradictory:
 296        return {
 297            "classification": "contradictory",
 298            "observation_ids": tuple(
 299                observation.event_id for observation in contradictory
 300            ),
 301        }
 302    if compatible:
 303        return {
 304            "classification": "compatible",
 305            "observation_ids": tuple(observation.event_id for observation in compatible),
 306        }
 307
 308    return {
 309        "classification": "inconclusive",
 310        "observation_ids": tuple(observation.event_id for observation in unique_observations),
 311    }

 314def aggregate_verification_responses(
 315    initial_observation: Observation,
 316    responses,
 317    conversation_id: str,
 318    request_message_id: str,
 319    max_distance: float,
 320    max_time_difference: float,
 321    *,
 322    incompatibility_rules: Mapping[str, object] | None = None,
 323    compatibility_rules: Mapping[str, object] | None = None,
 324    expected_incident_id: str | None = None,
 325) -> dict[str, object]:
 326    """Juntar as várias respostas a um pedido de verificação.
 327
 328    Só aceita mensagens ``VERIFY_RESPONSE`` da conversa esperada e que se
 329    refiram ao pedido esperado. Cada mensagem é processada uma vez (por
 330    ``message_id``) e cada observação contada uma vez (por ``event_id``). As
 331    ``source_ids`` e as fontes compatíveis vêm das próprias observações, nunca
 332    do remetente nem do ``source_id`` da resposta; só a ausência informativa
 333    (e, por isso, ``coverage_source_ids``) usa o ``source_id`` da resposta.
 334    Várias observações da mesma fonte contam como uma só fonte compatível.
 335
 336    Args:
 337        initial_observation (Observation): Observação que originou o pedido.
 338        responses (Iterable[Message]): ``VERIFY_RESPONSE`` recebidas.
 339        conversation_id (str): Conversa do pedido de verificação.
 340        request_message_id (str): ID do ``VERIFY_REQUEST`` a que as respostas respondem.
 341        max_distance (float): Distância máxima para observações compatíveis (inclusiva).
 342        max_time_difference (float): Diferença temporal máxima (inclusiva).
 343        incompatibility_rules (Mapping[str, object] | None): Tipos de evento que
 344            contradizem cada tipo.
 345        compatibility_rules (Mapping[str, object] | None): Tipos de evento, além do
 346            próprio, compatíveis com cada tipo.
 347        expected_incident_id (str | None): Incidente esperado; ``None`` não o verifica.
 348
 349    Returns:
 350        dict[str, object]: ``classification``, as ``observations`` e
 351            ``observation_ids`` independentes, os IDs compatíveis, contraditórios e em
 352            conflito, os ``source_ids`` distintos, ``compatible_source_ids``,
 353            ``compatible_source_count``, ``informative_absence_source_ids`` e
 354            ``coverage_source_ids``.
 355
 356    Raises:
 357        TypeError: Se ``initial_observation`` não for ``Observation``, se alguma resposta
 358            não for ``Message``, ou se um limite ou as regras tiverem o tipo errado.
 359        ValueError: Se um identificador estiver vazio, um limite for negativo ou não
 360            finito, ou se uma resposta for inválida ou pertencer a outra conversa,
 361            pedido ou incidente.
 362    """
 363    _validate_observation(initial_observation, "initial_observation")
 364    _validate_reference(conversation_id, "conversation_id")
 365    _validate_reference(request_message_id, "request_message_id")
 366    if expected_incident_id is not None:
 367        _validate_reference(expected_incident_id, "expected_incident_id")
 368    _validate_limit(max_distance, "max_distance")
 369    _validate_limit(max_time_difference, "max_time_difference")
 370
 371    seen_message_ids = set()
 372    observation_variants = {}
 373    observation_interpretations = {}
 374    compatible_ids = set()
 375    contradictory_ids = set()
 376    informative_absence_source_ids = set()
 377    saw_observation_response = False
 378    saw_no_data = False
 379
 380    for response in responses:
 381        if not isinstance(response, Message):
 382            raise TypeError("Each response must be a Message.")
 383        observations, status = _observations_from_response(response)
 384        if response.conversation_id != conversation_id:
 385            raise ValueError("VERIFY_RESPONSE belongs to another conversation.")
 386        if response.payload["request_message_id"] != request_message_id:
 387            raise ValueError("VERIFY_RESPONSE refers to another request.")
 388        if expected_incident_id is not None and response.incident_id != expected_incident_id:
 389            raise ValueError("VERIFY_RESPONSE belongs to another incident.")
 390        if response.message_id in seen_message_ids:
 391            continue
 392        seen_message_ids.add(response.message_id)
 393
 394        result = classify_verification_response(
 395            initial_observation,
 396            response,
 397            max_distance,
 398            max_time_difference,
 399            incompatibility_rules=incompatibility_rules,
 400            compatibility_rules=compatibility_rules,
 401        )
 402        if status == "no_data":
 403            saw_no_data = True
 404            continue
 405        if status == "informative_absence":
 406            if result["classification"] == "informative_absence":
 407                if response.source_id:
 408                    informative_absence_source_ids.add(response.source_id)
 409            continue
 410        saw_observation_response = True
 411
 412        for observation in observations:
 413            if (
 414                observation.event_id == initial_observation.event_id
 415                or observation.source_id == initial_observation.source_id
 416            ):
 417                continue
 418            variants = observation_variants.setdefault(observation.event_id, [])
 419            if observation not in variants:
 420                variants.append(observation)
 421
 422        # Os IDs de evidência mista são preservados mesmo quando a
 423        # classificação individual é inconclusive.
 424        compatible_ids.update(result.get("compatible_observation_ids", ()))
 425        contradictory_ids.update(result.get("contradictory_observation_ids", ()))
 426        if result["classification"] == "compatible":
 427            compatible_ids.update(result["observation_ids"])
 428        elif result["classification"] == "contradictory":
 429            contradictory_ids.update(result["observation_ids"])
 430
 431        for observation_id in result.get("compatible_observation_ids", ()):
 432            observation_interpretations.setdefault(observation_id, set()).add(
 433                "compatible"
 434            )
 435        for observation_id in result.get("contradictory_observation_ids", ()):
 436            observation_interpretations.setdefault(observation_id, set()).add(
 437                "contradictory"
 438            )
 439        if result["classification"] == "compatible":
 440            for observation_id in result["observation_ids"]:
 441                observation_interpretations.setdefault(observation_id, set()).add(
 442                    "compatible"
 443                )
 444        elif result["classification"] == "contradictory":
 445            for observation_id in result["observation_ids"]:
 446                observation_interpretations.setdefault(observation_id, set()).add(
 447                    "contradictory"
 448                )
 449
 450    unique_observations = {
 451        event_id: min(variants, key=_observation_sort_key)
 452        for event_id, variants in observation_variants.items()
 453    }
 454    conflicted_ids = {
 455        event_id
 456        for event_id, variants in observation_variants.items()
 457        if len(variants) > 1
 458        or len(observation_interpretations.get(event_id, set())) > 1
 459    }
 460    compatible_ids.difference_update(conflicted_ids)
 461    contradictory_ids.difference_update(conflicted_ids)
 462    compatible_ids.intersection_update(unique_observations)
 463    contradictory_ids.intersection_update(unique_observations)
 464    source_ids = {
 465        observation.source_id
 466        for variants in observation_variants.values()
 467        for observation in variants
 468    }
 469    compatible_source_ids = {
 470        unique_observations[event_id].source_id for event_id in compatible_ids
 471    }
 472
 473    if conflicted_ids or (compatible_ids and contradictory_ids):
 474        classification = "inconclusive"
 475    elif compatible_ids:
 476        classification = "compatible"
 477    elif contradictory_ids:
 478        classification = "contradictory"
 479    elif informative_absence_source_ids:
 480        classification = "informative_absence"
 481    elif saw_observation_response:
 482        classification = "inconclusive"
 483    elif saw_no_data or not seen_message_ids:
 484        classification = "no_data"
 485    else:  # Defensive fallback for future response statuses.
 486        classification = "inconclusive"
 487
 488    return {
 489        "classification": classification,
 490        "observations": tuple(model_to_dict(unique_observations[key]) for key in sorted(unique_observations)),
 491        "observation_ids": tuple(sorted(unique_observations)),
 492        "compatible_observation_ids": tuple(sorted(compatible_ids)),
 493        "contradictory_observation_ids": tuple(sorted(contradictory_ids)),
 494        "conflicted_observation_ids": tuple(sorted(conflicted_ids)),
 495        "source_ids": tuple(sorted(source_ids)),
 496        "compatible_source_ids": tuple(sorted(compatible_source_ids)),
 497        "compatible_source_count": len(compatible_source_ids),
 498        "informative_absence_source_ids": tuple(
 499            sorted(informative_absence_source_ids)
 500        ),
 501        "coverage_source_ids": tuple(
 502            sorted(source_ids | informative_absence_source_ids)
 503        ),
 504    }

 525def evaluate_reputation_corroboration(
 526    initial_observation: Observation,
 527    aggregate_summary: Mapping[str, object],
 528    eligible_source_ids,
 529    reputation_scores: Mapping[str, float],
 530    source_profiles: Mapping[str, Mapping[str, str]],
 531    max_distance: float,
 532    max_time_difference: float,
 533    *,
 534    default_reputation: float = 0.5,
 535) -> dict[str, object]:
 536    """Calcular o score heurístico de suspeita, ponderado pela reputação.
 537
 538    O score nunca é uma probabilidade calibrada.
 539    Peso = reputação × confiança declarada × proximidade temporal × espacial.
 540    Cada grupo de fontes correlacionadas contribui no máximo uma vez por sinal
 541    (positivo/negativo). Metadados desconhecidos partilham um grupo conservador.
 542    Ausência informativa reduz apoio; timeout/no_data não o reduz.
 543
 544    Args:
 545        initial_observation (Observation): Observação que originou o pedido.
 546        aggregate_summary (Mapping[str, object]): Resultado de
 547            :func:`aggregate_verification_responses`.
 548        eligible_source_ids (Iterable[str]): Fontes elegíveis, de
 549            :func:`select_eligible_sources`.
 550        reputation_scores (Mapping[str, float]): Reputação de cada fonte, entre 0 e 1.
 551        source_profiles (Mapping[str, Mapping[str, str]]): ``modality`` e
 552            ``correlation_group`` de cada fonte.
 553        max_distance (float): Distância máxima para evidência compatível (inclusiva).
 554        max_time_difference (float): Diferença temporal máxima (inclusiva).
 555        default_reputation (float): Reputação das fontes sem score, entre 0 e 1.
 556
 557    Returns:
 558        dict[str, object]: Indicadores de
 559            :func:`evaluate_preliminary_corroboration` acrescidos de
 560            ``suspicion_score`` (entre 0 e 1), ``support``, ``opposition``,
 561            ``diversity_factor``, ``supporting_group_count``, ``modalities``,
 562            ``conflicting_evidence`` e os ``factors`` de cada evidência.
 563
 564    Raises:
 565        TypeError: Se ``initial_observation`` não for ``Observation``, se
 566            ``aggregate_summary`` não for ``Mapping``, se um limite tiver o tipo errado
 567            ou se ``eligible_source_ids`` não for iterável.
 568        ValueError: Se um limite for negativo ou não finito, uma reputação estiver fora
 569            de [0, 1], se os perfis ou os IDs elegíveis forem inválidos, ou se uma
 570            observação do resumo for inválida.
 571    """
 572    _validate_observation(initial_observation, "initial_observation")
 573    _validate_limit(max_distance, "max_distance")
 574    _validate_limit(max_time_difference, "max_time_difference")
 575    for value in (default_reputation, *reputation_scores.values()):
 576        if not is_finite_number(value) or not 0 <= value <= 1:
 577            raise ValueError("Reputação deve estar entre 0 e 1.")
 578    validate_source_profiles(source_profiles)
 579    eligible = set(_unique_identifiers(eligible_source_ids, "eligible_source_ids"))
 580    evaluation = evaluate_preliminary_corroboration(aggregate_summary, eligible)
 581    positive = set(aggregate_summary.get("compatible_observation_ids", ()))
 582    negative = set(aggregate_summary.get("contradictory_observation_ids", ()))
 583    conflicted = set(aggregate_summary.get("conflicted_observation_ids", ()))
 584    factors = []
 585    groups = {}
 586
 587    def add(source_id, event_id, relation, confidence, temporal, spatial):
 588        """Registar um fator de evidência e atualizar o grupo de correlação da fonte.
 589
 590        Cada grupo guarda apenas o maior peso por relação. A modalidade só conta
 591        para a diversidade com apoio positivo, peso não nulo e modalidade conhecida.
 592        """
 593        profile = source_profiles.get(source_id, {})
 594        group = profile.get("correlation_group", "unknown")
 595        modality = profile.get("modality", "unknown")
 596        reputation = reputation_scores.get(source_id, default_reputation)
 597        weight = reputation * confidence * temporal * spatial
 598        factors.append({
 599            "source_id": source_id, "event_id": event_id, "relation": relation,
 600            "reputation": reputation, "confidence": confidence,
 601            "temporal_factor": temporal, "spatial_factor": spatial,
 602            "modality": modality, "correlation_group": group, "weight": weight,
 603        })
 604        entry = groups.setdefault(group, {"positive": 0.0, "negative": 0.0, "modalities": set()})
 605        entry[relation] = max(entry[relation], weight)
 606        if relation == "positive" and weight > 0 and modality != "unknown":
 607            entry["modalities"].add(modality)
 608
 609    add(initial_observation.source_id, initial_observation.event_id, "positive",
 610        initial_observation.confidence_score, 1.0, 1.0)
 611    for raw in aggregate_summary.get("observations", ()):
 612        observation = model_from_dict(raw, Observation)
 613        event_id = observation.event_id
 614        if (observation.source_id not in eligible or event_id in conflicted
 615                or observation.source_id == initial_observation.source_id):
 616            continue
 617        if event_id not in positive | negative or not are_observations_compatible(
 618            initial_observation, observation, max_distance, max_time_difference
 619        ):
 620            continue
 621        distance = hypot(initial_observation.position.x - observation.position.x,
 622                         initial_observation.position.y - observation.position.y)
 623        delta = abs(initial_observation.observed_at - observation.observed_at)
 624        temporal = 1 / (1 + delta / max_time_difference) if max_time_difference else 1.0
 625        spatial = 1 / (1 + distance / max_distance) if max_distance else 1.0
 626        add(observation.source_id, event_id, "negative" if event_id in negative else "positive",
 627            observation.confidence_score, temporal, spatial)
 628    for source_id in sorted(set(aggregate_summary.get("informative_absence_source_ids", ())) & eligible):
 629        if source_id != initial_observation.source_id:
 630            add(source_id, None, "negative", 1.0, 1.0, 1.0)
 631
 632    supporting = [entry for entry in groups.values() if entry["positive"] > entry["negative"]]
 633    modalities = set().union(*(entry["modalities"] for entry in supporting))
 634    diversity = 0.5 + 0.25 * min(2, len(modalities))
 635    support = fsum(entry["positive"] for entry in groups.values())
 636    opposition = fsum(entry["negative"] for entry in groups.values())
 637    # ponytail: máximo por grupo é uma heurística; substituir só após medir dependências.
 638    # Duas unidades de apoio líquido saturam a escala. Evidência de peso zero
 639    # não dilui apoio existente só por acrescentar outro grupo.
 640    score = max(0.0, min(1.0, (support - opposition) / 2)) * diversity
 641    evaluation.update({
 642        "suspicion_score": score, "score_kind": "heuristic_not_probability",
 643        "support": support, "opposition": opposition, "diversity_factor": diversity,
 644        "supporting_group_count": len(supporting), "modalities": tuple(sorted(modalities)),
 645        "conflicting_evidence": bool(conflicted) or (support > 0 and opposition > 0),
 646        "factors": tuple(sorted(factors, key=lambda item: (item["source_id"], item["event_id"] or ""))),
 647        "group_count": len(groups),
 648        "max_distance": max_distance, "max_time_difference": max_time_difference,
 649    })
 650    return evaluation

 667def evaluate_preliminary_corroboration(
 668    aggregate_summary: Mapping[str, object],
 669    eligible_source_ids,
 670) -> dict[str, object]:
 671    """Calcular os indicadores de cobertura a partir do resumo agregado.
 672
 673    ``eligible_source_ids`` tem de vir de :func:`select_eligible_sources`. Só
 674    conta as fontes, sem repetições, que já estão no resumo:
 675
 676    - **score preliminar** = fontes elegíveis compatíveis / fontes elegíveis;
 677    - **cobertura de informação** = fontes elegíveis com observações ou ausência
 678      informativa válida / fontes elegíveis.
 679
 680    São indicadores descritivos, não probabilidades calibradas: fontes
 681    diferentes podem partilhar hardware, local, software ou pontos cegos, por
 682    isso não são necessariamente independentes.
 683
 684    Args:
 685        aggregate_summary (Mapping[str, object]): Resultado de
 686            :func:`aggregate_verification_responses`.
 687        eligible_source_ids (Iterable[str]): Fontes elegíveis para a observação.
 688
 689    Returns:
 690        dict[str, object]: ``classification``, ``compatible_observation_ids``,
 691            ``eligible_source_count``, IDs e contagens das fontes compatíveis e
 692            observadas, ``preliminary_score`` e ``information_coverage`` (ambos
 693            ``None`` se não houver fontes elegíveis).
 694
 695    Raises:
 696        TypeError: Se ``aggregate_summary`` não for ``Mapping`` ou um conjunto de IDs
 697            não for iterável.
 698        ValueError: Se algum ID não for uma string não vazia.
 699    """
 700    if not isinstance(aggregate_summary, Mapping):
 701        raise TypeError("aggregate_summary must be a mapping.")
 702    eligible = _unique_identifiers(eligible_source_ids, "eligible_source_ids")
 703    eligible_set = set(eligible)
 704
 705    compatible_source_ids = _summary_identifiers(
 706        aggregate_summary, "compatible_source_ids"
 707    ) & eligible_set
 708    observed_source_ids = (
 709        _summary_identifiers(
 710            aggregate_summary,
 711            "coverage_source_ids",
 712        )
 713        or _summary_identifiers(aggregate_summary, "source_ids")
 714    ) & eligible_set
 715
 716    eligible_count = len(eligible_set)
 717    if not eligible_count:
 718        score = None
 719        coverage = None
 720        compatible_source_ids = set()
 721        observed_source_ids = set()
 722    else:
 723        score = len(compatible_source_ids) / eligible_count
 724        coverage = len(observed_source_ids) / eligible_count
 725
 726    return {
 727        "classification": aggregate_summary.get("classification"),
 728        "compatible_observation_ids": tuple(sorted(_summary_identifiers(
 729            aggregate_summary, "compatible_observation_ids"
 730        ))),
 731        "eligible_source_count": eligible_count,
 732        "compatible_source_ids": tuple(sorted(compatible_source_ids)),
 733        "compatible_source_count": len(compatible_source_ids),
 734        "observed_source_ids": tuple(sorted(observed_source_ids)),
 735        "observed_source_count": len(observed_source_ids),
 736        "preliminary_score": score,
 737        "information_coverage": coverage,
 738    }

 741def decide_preliminary_verification(
 742    evaluation_summary: Mapping[str, object],
 743    minimum_compatible_sources: int,
 744) -> dict[str, object]:
 745    """Escolher entre continuar a observar e pedir verificação por patrulha.
 746
 747    Regra operacional explícita: pede verificação quando o resumo atinge
 748    ``minimum_compatible_sources`` fontes compatíveis distintas e há fontes
 749    elegíveis; senão continua a observar. O limiar é um parâmetro de engenharia dado pelo chamador, não
 750    uma estimativa validada da probabilidade de caça furtiva.
 751
 752    Não processa mensagens nem observações: recebe o resultado de
 753    :func:`evaluate_preliminary_corroboration`, que já tirou as repetições.
 754
 755    Args:
 756        evaluation_summary (Mapping[str, object]): Resultado de
 757            :func:`evaluate_preliminary_corroboration`.
 758        minimum_compatible_sources (int): Fontes compatíveis distintas necessárias para
 759            pedir patrulha; inteiro positivo.
 760
 761    Returns:
 762        dict[str, object]: ``decision`` (``observe`` ou ``request_mobile_verification``),
 763            ``reason`` e os indicadores do resumo usados na decisão.
 764
 765    Raises:
 766        TypeError: Se ``evaluation_summary`` não for ``Mapping``, se
 767            ``minimum_compatible_sources`` não for inteiro, ou se um conjunto de IDs não
 768            for iterável.
 769        ValueError: Se ``minimum_compatible_sources`` não for positivo, se uma contagem
 770            não for um inteiro não negativo, ou se algum ID for inválido.
 771    """
 772    if not isinstance(evaluation_summary, Mapping):
 773        raise TypeError("evaluation_summary must be a mapping.")
 774    if isinstance(minimum_compatible_sources, bool) or not isinstance(
 775        minimum_compatible_sources, int
 776    ):
 777        raise TypeError("minimum_compatible_sources must be a positive integer.")
 778    if minimum_compatible_sources <= 0:
 779        raise ValueError("minimum_compatible_sources must be a positive integer.")
 780
 781    eligible_count = _nonnegative_count(
 782        evaluation_summary.get("eligible_source_count", 0),
 783        "eligible_source_count",
 784    )
 785    observed_count = _nonnegative_count(
 786        evaluation_summary.get("observed_source_count", 0),
 787        "observed_source_count",
 788    )
 789    compatible_source_ids = _summary_identifiers(
 790        evaluation_summary, "compatible_source_ids"
 791    )
 792    compatible_count = len(compatible_source_ids)
 793
 794    if compatible_count >= minimum_compatible_sources and eligible_count > 0:
 795        decision = "request_mobile_verification"
 796        reason = "compatible_source_threshold_reached"
 797    elif eligible_count == 0:
 798        decision = "observe"
 799        reason = "no_eligible_sources"
 800    elif observed_count == 0:
 801        decision = "observe"
 802        reason = "no_information"
 803    elif compatible_count == 0:
 804        decision = "observe"
 805        reason = "no_compatible_evidence"
 806    else:
 807        decision = "observe"
 808        reason = "insufficient_compatible_evidence"
 809
 810    return {
 811        "decision": decision,
 812        "reason": reason,
 813        "compatible_observation_ids": tuple(
 814            sorted(
 815                _summary_identifiers(
 816                    evaluation_summary, "compatible_observation_ids"
 817                )
 818            )
 819        ),
 820        "compatible_source_ids": tuple(sorted(compatible_source_ids)),
 821        "compatible_source_count": compatible_count,
 822        "eligible_source_count": eligible_count,
 823        "observed_source_count": observed_count,
 824        "preliminary_score": evaluation_summary.get("preliminary_score"),
 825        "information_coverage": evaluation_summary.get("information_coverage"),
 826        "minimum_compatible_sources": minimum_compatible_sources,
 827    }

 830def _observations_from_response(response: Message) -> tuple[tuple[Observation, ...], str]:
 831    """Validar um ``VERIFY_RESPONSE`` e devolver as suas observações e o ``status``.
 832
 833    Raises:
 834        TypeError: Se ``response`` não for ``Message``.
 835        ValueError: Se o tipo, o payload, o estado ou alguma observação forem inválidos,
 836            se o estado não for coerente com a lista de observações, ou se uma
 837            observação ou o fim do intervalo de ausência forem posteriores à emissão
 838            da resposta.
 839    """
 840    if not isinstance(response, Message):
 841        raise TypeError("response must be a Message.")
 842    if response.message_type is not MessageType.VERIFY_RESPONSE:
 843        raise ValueError("response must be a VERIFY_RESPONSE message.")
 844
 845    payload = response.payload
 846    if not isinstance(payload, dict):
 847        raise ValueError("VERIFY_RESPONSE payload must be an object.")
 848    request_message_id = payload.get("request_message_id")
 849    if not isinstance(request_message_id, str) or not request_message_id.strip():
 850        raise ValueError("VERIFY_RESPONSE must identify its request message.")
 851
 852    status = payload.get("status")
 853    if status not in {
 854        "observations_available",
 855        "no_data",
 856        "informative_absence",
 857    }:
 858        raise ValueError("VERIFY_RESPONSE status is invalid.")
 859    raw_observations = payload.get("observations")
 860    if not isinstance(raw_observations, list):
 861        raise ValueError("VERIFY_RESPONSE observations must be a list.")
 862    if status in {"no_data", "informative_absence"} and raw_observations:
 863        raise ValueError("A no_data response cannot contain observations.")
 864    if status == "observations_available" and not raw_observations:
 865        raise ValueError("An observations_available response must contain observations.")
 866    if status == "informative_absence":
 867        absence = payload.get("absence")
 868        observed_until = absence.get("observed_until") if isinstance(absence, Mapping) else None
 869        if is_finite_number(observed_until) and observed_until > response.created_at:
 870            raise ValueError("VERIFY_RESPONSE cannot claim a future observation interval.")
 871
 872    observations = []
 873    for raw_observation in raw_observations:
 874        try:
 875            observation = model_from_dict(raw_observation, Observation)
 876        except (TypeError, ValueError, KeyError) as error:
 877            raise ValueError("VERIFY_RESPONSE contains an invalid observation.") from error
 878        if observation.observed_at > response.created_at:
 879            raise ValueError("VERIFY_RESPONSE cannot contain a future observation.")
 880        observations.append(observation)
 881    return tuple(observations), status

 933def _hypothesis_is_correlated(
 934    initial_observation: Observation,
 935    response: Message,
 936) -> bool:
 937    """Verificar se a hipótese ecoada na resposta corresponde à observação inicial.
 938
 939    Exige ID não vazio, o mesmo tipo de evento, a mesma posição e um intervalo
 940    de validade finito que contenha o instante da observação. Hipótese ausente
 941    ou malformada devolve ``False`` em vez de levantar erro.
 942    """
 943    hypothesis = response.payload.get("hypothesis")
 944    if not isinstance(hypothesis, Mapping):
 945        return False
 946    try:
 947        hypothesis_id = hypothesis["hypothesis_id"]
 948        event_type = hypothesis["event_type"]
 949        position = model_from_dict(hypothesis["position"], Position)
 950        valid_from = hypothesis["valid_from"]
 951        valid_until = hypothesis["valid_until"]
 952    except (KeyError, TypeError, ValueError):
 953        return False
 954    return (
 955        isinstance(hypothesis_id, str)
 956        and bool(hypothesis_id.strip())
 957        and event_type == initial_observation.event_type
 958        and position == initial_observation.position
 959        and is_finite_number(valid_from)
 960        and is_finite_number(valid_until)
 961        and valid_from <= initial_observation.observed_at <= valid_until
 962        and valid_from <= valid_until
 963    )

 966def _informative_absence_is_valid(
 967    initial_observation: Observation,
 968    response: Message,
 969) -> bool:
 970    """Verificar se uma ausência declarada conta como evidência negativa.
 971
 972    Exige fonte identificada, hipótese correlacionada e referida pela ausência,
 973    não-deteção explícita, capacidade adequada ao sinal e validada, oportunidade
 974    confirmada, intervalo de observação que cubra o da hipótese e posição da
 975    hipótese dentro do raio de cobertura. Campos em falta devolvem ``False``.
 976    """
 977    hypothesis = response.payload.get("hypothesis")
 978    absence = response.payload.get("absence")
 979    if not isinstance(hypothesis, Mapping) or not isinstance(absence, Mapping):
 980        return False
 981    if not isinstance(response.source_id, str) or not response.source_id.strip():
 982        return False
 983    if not _hypothesis_is_correlated(initial_observation, response):
 984        return False
 985    try:
 986        coverage = absence["coverage"]
 987        coverage_position = model_from_dict(coverage["position"], Position)
 988        radius = coverage["radius"]
 989        observed_from = absence["observed_from"]
 990        observed_until = absence["observed_until"]
 991        capability = coverage["capability"]
 992        capability_validated = coverage["capability_validated"]
 993        opportunity_confirmed = coverage["opportunity_confirmed"]
 994        hypothesis_id = absence["hypothesis_id"]
 995        declared_non_detection = absence["declared_non_detection"]
 996        hypothesis_from = hypothesis["valid_from"]
 997        hypothesis_until = hypothesis["valid_until"]
 998        hypothesis_position = model_from_dict(hypothesis["position"], Position)
 999    except (KeyError, TypeError, ValueError):
1000        return False
1001    if (
1002        not isinstance(coverage, Mapping)
1003        or not isinstance(hypothesis_id, str)
1004        or not hypothesis_id.strip()
1005        or hypothesis_id != hypothesis.get("hypothesis_id")
1006        or declared_non_detection is not True
1007        or capability_validated is not True
1008        or opportunity_confirmed is not True
1009        or not capability_supports_signal(capability, hypothesis.get("event_type"))
1010        or not is_finite_number(radius)
1011        or radius < 0
1012        or not is_finite_number(observed_from)
1013        or not is_finite_number(observed_until)
1014        or observed_from > observed_until
1015        or observed_from > hypothesis_from
1016        or observed_until < hypothesis_until
1017    ):
1018        return False
1019    return positions_within_distance(coverage_position, hypothesis_position, radius)

555        max_time_difference (float): Diferença temporal máxima (inclusiva).
556        default_reputation (float): Reputação das fontes sem score, entre 0 e 1.

563            ``conflicting_evidence`` e os ``factors`` de cada evidência.

567            ``aggregate_summary`` não for ``Mapping``, se um limite tiver o tipo errado

572    """
573    _validate_observation(initial_observation, "initial_observation")
574    _validate_limit(max_distance, "max_distance")
575    _validate_limit(max_time_difference, "max_time_difference")
576    for value in (default_reputation, *reputation_scores.values()):
577        if not is_finite_number(value) or not 0 <= value <= 1:
578            raise ValueError("Reputação deve estar entre 0 e 1.")
579    validate_source_profiles(source_profiles)
580    eligible = set(_unique_identifiers(eligible_source_ids, "eligible_source_ids"))

582    positive = set(aggregate_summary.get("compatible_observation_ids", ()))
583    negative = set(aggregate_summary.get("contradictory_observation_ids", ()))
584    conflicted = set(aggregate_summary.get("conflicted_observation_ids", ()))
585    factors = []
586    groups = {}
587
588    def add(source_id, event_id, relation, confidence, temporal, spatial):
589        """Registar um fator de evidência e atualizar o grupo de correlação da fonte.
590
591        Cada grupo guarda apenas o maior peso por relação. A modalidade só conta
592        para a diversidade com apoio positivo, peso não nulo e modalidade conhecida.
593        """
594        profile = source_profiles.get(source_id, {})
595        group = profile.get("correlation_group", "unknown")
596        modality = profile.get("modality", "unknown")
597        reputation = reputation_scores.get(source_id, default_reputation)
598        weight = reputation * confidence * temporal * spatial
599        factors.append({
600            "source_id": source_id, "event_id": event_id, "relation": relation,
601            "reputation": reputation, "confidence": confidence,
602            "temporal_factor": temporal, "spatial_factor": spatial,
603            "modality": modality, "correlation_group": group, "weight": weight,
604        })
605        entry = groups.setdefault(group, {"positive": 0.0, "negative": 0.0, "modalities": set()})
606        entry[relation] = max(entry[relation], weight)
607        if relation == "positive" and weight > 0 and modality != "unknown":
608            entry["modalities"].add(modality)
609

613        observation = model_from_dict(raw, Observation)
614        event_id = observation.event_id
615        if (observation.source_id not in eligible or event_id in conflicted
616                or observation.source_id == initial_observation.source_id):
617            continue
618        if event_id not in positive | negative or not are_observations_compatible(
619            initial_observation, observation, max_distance, max_time_difference
620        ):
621            continue

623                         initial_observation.position.y - observation.position.y)
624        delta = abs(initial_observation.observed_at - observation.observed_at)
625        temporal = 1 / (1 + delta / max_time_difference) if max_time_difference else 1.0
626        spatial = 1 / (1 + distance / max_distance) if max_distance else 1.0
627        add(observation.source_id, event_id, "negative" if event_id in negative else "positive",
628            observation.confidence_score, temporal, spatial)
629    for source_id in sorted(set(aggregate_summary.get("informative_absence_source_ids", ())) & eligible):
630        if source_id != initial_observation.source_id:
631            add(source_id, None, "negative", 1.0, 1.0, 1.0)
632
633    supporting = [entry for entry in groups.values() if entry["positive"] > entry["negative"]]
634    modalities = set().union(*(entry["modalities"] for entry in supporting))
635    diversity = 0.5 + 0.25 * min(2, len(modalities))
636    support = fsum(entry["positive"] for entry in groups.values())
637    opposition = fsum(entry["negative"] for entry in groups.values())
638    # ponytail: máximo por grupo é uma heurística; substituir só após medir dependências.
639    # Duas unidades de apoio líquido saturam a escala. Evidência de peso zero
640    # não dilui apoio existente só por acrescentar outro grupo.
641    score = max(0.0, min(1.0, (support - opposition) / 2)) * diversity
642    evaluation.update({
643        "suspicion_score": score, "score_kind": "heuristic_not_probability",
644        "support": support, "opposition": opposition, "diversity_factor": diversity,
645        "supporting_group_count": len(supporting), "modalities": tuple(sorted(modalities)),
646        "conflicting_evidence": bool(conflicted) or (support > 0 and opposition > 0),
647        "factors": tuple(sorted(factors, key=lambda item: (item["source_id"], item["event_id"] or ""))),
648        "group_count": len(groups),
649        "max_distance": max_distance, "max_time_difference": max_time_difference,
650    })
651    return evaluation
