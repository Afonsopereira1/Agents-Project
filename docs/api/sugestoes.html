Quatro grupos, cada um varre um só parâmetro: guardas cenário E (16 ameaças) com 2, 3 e 4 guardas reenvios perda de 10, 20 e 30 % com 0 a 4 reenvios por pedido (cenário A, 20 sementes) distancia ameaça de 15 s e o guarda a 5, 6, 7, 8 e 9 células do alvo (fontes sem ruído) cobertura ameaça em (7,0), fora do alcance do sensor-1, sem e com um sensor iniciador que a alcança Grava results/stress/sugestoes.csv. Uso, a partir da raiz: PYTHONPATH=src python scripts/sugestoes.py [grupo ...]

 31def row(grupo, valor, reports):
 32    """Agregar relatórios numa linha do CSV, contando uma ocorrência por iniciador.
 33
 34    Args:
 35        grupo (str): Nome do grupo, na coluna ``grupo``.
 36        valor (str): Descrição do caso, na coluna ``valor``.
 37        reports (list[dict]): Relatórios das execuções do caso (saída de ``run_mvp``).
 38
 39    Returns:
 40        dict: Linha com as colunas de ``FIELDS``; ``t_fecho_sim_mean`` é a média, em s
 41            simulados, das ocorrências fechadas com ameaça confirmada.
 42    """
 43    views = [v for r in reports for v in incidents(r)]
 44    done = [v for v in views if v["status"] == "closed" and v["reason"] == "confirmed"]
 45    return {"grupo": grupo, "valor": valor, "runs": len(reports),
 46            "ameacas": sum(e["is_threat"] for r in reports for e in r["ground_truth"]["events"]),
 47            "com_alerta": sum(v["status"] is not None for v in views), "chegam_ao_guarda": len(done),
 48            "unresolved": sum(v["status"] == "unresolved" for v in views),
 49            "t_fecho_sim_mean": mean(v["closed_at"] for v in done),
 50            "msgs_mean": mean(len(r["messages"]) for r in reports)}

 53async def guardas():
 54    """Testar o cenário E com 2, 3 e 4 guardas (acrescenta ``ranger-3`` e ``ranger-4``).
 55
 56    Returns:
 57        list[dict]: Uma linha por caso, com as colunas de ``FIELDS``: ``grupo``, ``valor``,
 58            ``runs``, ``ameacas``, ``com_alerta``, ``chegam_ao_guarda``, ``unresolved``,
 59            ``t_fecho_sim_mean`` (média, por execução, do instante do último fecho, em s
 60            simulados) e ``msgs_mean``.
 61    """
 62    extra = {"ranger-3": {"x": 8, "y": 3}, "ranger-4": {"x": 8, "y": 9}}
 63    out = []
 64    for n in (2, 3, 4):
 65        config = copy.deepcopy(E)
 66        config["world"]["rangers"].update(dict(list(extra.items())[:n - 2]))
 67        rows = await run_scenario(config, "full")
 68        out.append({"grupo": "guardas", "valor": f"{n} guardas para 16 ameaças", "runs": len(rows),
 69                    "ameacas": 16 * len(rows), "com_alerta": sum(r["incidents"] for r in rows),
 70                    # Em E todas as fechadas são confirmadas pelo guarda (ver scripts/scenario_e.py).
 71                    "chegam_ao_guarda": sum(r["incidents_closed"] for r in rows),
 72                    "unresolved": sum(r["incidents_unresolved"] for r in rows),
 73                    "t_fecho_sim_mean": mean(r["last_close_time"] for r in rows),
 74                    "msgs_mean": mean(r["messages_sent"] for r in rows)})
 75    return out

 78async def reenvios():
 79    """Testar perdas de 10, 20 e 30 % com 0 a 4 reenvios; 20 execuções por caso no cenário A.
 80
 81    Returns:
 82        list[dict]: Uma linha por caso, com as colunas de ``FIELDS``: ``grupo``, ``valor``,
 83            ``runs``, ``ameacas``, ``com_alerta``, ``chegam_ao_guarda``, ``unresolved``,
 84            ``t_fecho_sim_mean`` (sempre ``None``) e ``msgs_mean``.
 85    """
 86    out = []
 87    for loss in (0.1, 0.2, 0.3):
 88        for n in range(5):
 89            r = await case(f"perda {loss:.0%}, {n} reenvios", A["resources"], {"loss": loss}, 20,
 90                           {"retransmit": n, "max_seconds": None})
 91            out.append({"grupo": "reenvios", "valor": r["caso"], "runs": r["runs"], "ameacas": r["runs"],
 92                        "com_alerta": r["alerta"], "chegam_ao_guarda": r["closed"], "unresolved": r["unresolved"],
 93                        "t_fecho_sim_mean": None, "msgs_mean": r["msgs_mean"]})
 94    return out

 97async def simple(run_id, events, resources, sources, settings, **options):
 98    """Correr uma execução com reputação nova; a semente é 1000 + o sufixo do ``run_id``.
 99
100    Args:
101        run_id (str): Id da execução, terminado em ``-N`` (ex.: ``dist-3``).
102        events (list[dict]): Eventos do episódio.
103        resources (dict): Overrides dos recursos.
104        sources (dict): Fontes (tipo e alcance) por ID de sensor.
105        settings (dict): ``response_settings`` do cenário, com ``reputation``.
106        **options: Argumentos extra do ``run_mvp`` (ex.: ``extra_initiators``); sobrepõem-se às
107            condições publicadas.
108
109    Returns:
110        dict: Relatório da execução devolvido pelo ``run_mvp``.
111    """
112    return await run_mvp(response_settings=settings, reputation=ReputationBook(**settings["reputation"]),
113                         **{**PUBLISHED_CONDITIONS, **options}, max_seconds=30.0,
114                         scenario={"run_id": run_id, "seed": 1000 + int(run_id.rsplit("-", 1)[1]),
115                                   "sources": sources, "events": events, "resources": resources})

118async def distancia():
119    """Testar a distância do guarda até ao alvo, de 5 a 9 células, com uma ameaça de 15 s.
120
121    O veículo verifica aos 6 s, o guarda sai depois e demora distância + 1 s. Fontes sem
122    ruído; 5 execuções por caso.
123
124    Returns:
125        list[dict]: Uma linha por caso, com as colunas de ``FIELDS``: ``grupo``, ``valor``,
126            ``runs``, ``ameacas``, ``com_alerta``, ``chegam_ao_guarda``, ``unresolved``,
127            ``t_fecho_sim_mean`` e ``msgs_mean``.
128    """
129    sources = {"sensor-1": {"kind": "acoustic", "detection_range": 5.0}, "sensor-2": {"kind": "visual", "detection_range": 5.0}}
130    out = []
131    for cells, (x, y) in ((5, (5, 5)), (6, (6, 5)), (7, (7, 5)), (8, (7, 6)), (9, (7, 7))):
132        reports = [await simple(f"dist-{rep}", [{**THREAT, "duration": 15.0}],
133                                {"drone": {"load": 0.5}, "ranger": {"position": {"x": x, "y": y}}},
134                                sources, copy.deepcopy(A["response_settings"])) for rep in range(5)]
135        out.append(row("distancia", f"guarda a {cells} células, ameaça de 15 s", reports))
136    return out

139async def cobertura():
140    """Testar uma ameaça em (7,0) sem e com um iniciador adicional em (6,1); 10 execuções cada.
141
142    Returns:
143        list[dict]: Uma linha por caso, com as colunas de ``FIELDS``: ``grupo``, ``valor``,
144            ``runs``, ``ameacas``, ``com_alerta``, ``chegam_ao_guarda``, ``unresolved``,
145            ``t_fecho_sim_mean`` e ``msgs_mean``.
146    """
147    out = []
148    for label, extra in (("só o sensor-1 inicia", None), ("mais um iniciador em (6,1)", {"sensor-3": Position(6, 1)})):
149        settings = copy.deepcopy(A["response_settings"])
150        sources = dict(A["sources"])
151        options = {}
152        if extra:
153            sources["sensor-3"] = A["sources"]["sensor-1"]
154            settings["sources"]["sensor-3"] = {"modality": "acoustic", "correlation_group": "local_sensor-3"}
155            options = {"extra_initiators": extra, "extra_informants": {}}  # {}: corrobora com o sensor-2
156        reports = [await simple(f"cob-{rep}", [{**THREAT, "position": {"x": 7, "y": 0}}], A["resources"],
157                                sources, settings, **options) for rep in range(10)]
158        out.append(row("cobertura", f"ameaça em (7,0), {label}", reports))
159    return out
