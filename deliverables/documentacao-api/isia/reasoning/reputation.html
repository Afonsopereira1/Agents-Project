 10class EvaluationResult:
 11    """Adjudicação explícita; não é uma votação nem uma mensagem MISSION_RESULT.
 12
 13    O avaliador associa as observações ao incidente e verifica a proveniência.
 14    ``refuted`` exige exclusão fundamentada da ameaça no local/intervalo das
 15    observações; uma patrulha que não encontrou nada não basta.
 16
 17    Attributes:
 18        result_id (str): Identificador do resultado.
 19        incident_id (str): Incidente avaliado.
 20        observation_ids (tuple[str, ...]): Observações avaliadas (pelo menos uma, sem repetições).
 21        outcome (str): ``confirmed``, ``refuted``, ``inconclusive`` ou ``failed``.
 22        basis (str): Fundamento: ``ground_truth``, ``independent_mission`` ou ``consensus``.
 23        evaluated_at (float): Instante da avaliação, em segundos simulados.
 24        evidence_ids (tuple[str, ...]): Mensagens ou registos usados como evidência.
 25        evidence_source_ids (tuple[str, ...]): Fontes que produziram a evidência.
 26        final (bool): Se o resultado já não pode mudar.
 27        sufficient_coverage (bool): Se a evidência cobre o local e o intervalo das observações.
 28    """
 29
 30    result_id: str
 31    incident_id: str
 32    observation_ids: tuple[str, ...]
 33    outcome: str
 34    basis: str
 35    evaluated_at: float
 36    evidence_ids: tuple[str, ...] = ()
 37    evidence_source_ids: tuple[str, ...] = ()
 38    final: bool = False
 39    sufficient_coverage: bool = False
 40
 41    def __post_init__(self):
 42        """Validar identificadores, referências, resultado, fundamento e instante.
 43
 44        As referências têm de ser tuplos de IDs únicos e tem de haver pelo menos
 45        uma observação avaliada.
 46
 47        Raises:
 48            ValueError: Se algum campo for inválido.
 49        """
 50        for value in (self.result_id, self.incident_id):
 51            if not isinstance(value, str) or not value.strip():
 52                raise ValueError("Resultado e incidente precisam de identificadores.")
 53        for values in (self.observation_ids, self.evidence_ids, self.evidence_source_ids):
 54            if not isinstance(values, tuple) or any(
 55                not isinstance(value, str) or not value.strip() for value in values
 56            ) or len(values) != len(set(values)):
 57                raise ValueError("Referências devem ser tuplos de IDs únicos.")
 58        if not self.observation_ids:
 59            raise ValueError("O resultado deve identificar observações avaliadas.")
 60        if self.outcome not in ("confirmed", "refuted", "inconclusive", "failed"):
 61            raise ValueError("Resultado de avaliação desconhecido.")
 62        if self.basis not in ("ground_truth", "independent_mission", "consensus"):
 63            raise ValueError("Fundamento de avaliação desconhecido.")
 64        if not is_finite_number(self.evaluated_at) or self.evaluated_at < 0:
 65            raise ValueError("evaluated_at deve ser finito e não negativo.")
 66        if type(self.final) is not bool or type(self.sufficient_coverage) is not bool:
 67            raise ValueError("final e sufficient_coverage devem ser booleanos.")
 68
 69    def rejection_reason(self, observation: Observation, incident_id: str) -> str | None:
 70        """Devolver o motivo que impede este resultado de atualizar a reputação.
 71
 72        É aceite um resultado final ``confirmed`` ou ``refuted`` deste incidente e
 73        observação, não anterior a ela, com cobertura suficiente e evidência que não
 74        seja consenso, nem circular, nem (em missão independente) proveniente da
 75        própria fonte avaliada.
 76
 77        Args:
 78            observation (Observation): Observação cuja fonte seria atualizada.
 79            incident_id (str): Incidente a que a observação pertence.
 80
 81        Returns:
 82            str | None: Motivo da rejeição (ex.: ``unrelated_result``,
 83                ``circular_evidence``), ou ``None`` se o resultado for aceite.
 84        """
 85        if self.incident_id != incident_id or observation.event_id not in self.observation_ids:
 86            return "unrelated_result"
 87        if self.evaluated_at < observation.observed_at:
 88            return "result_precedes_observation"
 89        if not self.final or self.outcome not in ("confirmed", "refuted"):
 90            return "unresolved_result"
 91        if not self.sufficient_coverage or not self.evidence_ids:
 92            return "insufficient_reference"
 93        if self.basis == "consensus":
 94            return "agreement_is_not_reference"
 95        if set(self.evidence_ids) & set(self.observation_ids):
 96            return "circular_evidence"
 97        if self.basis == "independent_mission" and (
 98            not self.evidence_source_ids or observation.source_id in self.evidence_source_ids
 99        ):
100            return "non_independent_reference"
101        return None

104class ReputationBook:
105    """Uma atualização por fonte/incidente, com prior Beta e auditoria local.
106
107    Mede utilidade do sinal como alerta de ameaça, não exatidão física do sensor.
108    Um som real de trovão pode, portanto, ser um alerta de ameaça incorreto.
109    ``run_id`` distingue amostras de execuções identificadas pelo avaliador;
110    repetir o mesmo âmbito conserva a idempotência, sem presumir independência.
111
112    Attributes:
113        alpha (float): Parâmetro ``alpha`` do prior Beta, comum a todas as fontes.
114        beta (float): Parâmetro ``beta`` do prior Beta, comum a todas as fontes.
115        history (list[dict]): Registos de auditoria de todas as tentativas de atualização.
116    """
117
118    def __init__(self, alpha: float = 2.0, beta: float = 2.0):
119        """Inicializar o livro com o prior Beta comum a todas as fontes e sem histórico.
120
121        Args:
122            alpha (float): Pseudo-contagem inicial de acertos; finita e positiva.
123            beta (float): Pseudo-contagem inicial de erros; finita e positiva.
124
125        Raises:
126            ValueError: Se ``alpha`` ou ``beta`` não forem finitos e positivos, ou se a
127                sua soma não for finita.
128        """
129        if any(not is_finite_number(value) or value <= 0 for value in (alpha, beta)):
130            raise ValueError("alpha e beta devem ser finitos e positivos.")
131        if not is_finite_number(alpha + beta):
132            raise ValueError("A soma do prior deve ser finita.")
133        self.alpha, self.beta = alpha, beta
134        self._counts: dict[str, tuple[int, int]] = {}
135        self._evaluated_events: set[tuple[str | None, str]] = set()
136        self._evaluated_incidents: set[tuple[str | None, str, str]] = set()
137        self.history: list[dict] = []
138
139    def score(self, source_id: str) -> float:
140        """Devolver a média posterior Beta da fonte; sem avaliações, a média do prior.
141
142        Args:
143            source_id (str): Fonte a consultar.
144
145        Returns:
146            float: Reputação em ]0, 1[.
147        """
148        correct, incorrect = self._counts.get(source_id, (0, 0))
149        return (correct + self.alpha) / (correct + incorrect + self.alpha + self.beta)
150
151    def snapshot(self, source_ids) -> dict[str, float]:
152        """Devolver os scores das fontes dadas, sem duplicados e ordenados por identificador.
153
154        Args:
155            source_ids (Iterable[str]): Fontes a incluir; repetições são ignoradas.
156
157        Returns:
158            dict[str, float]: Reputação de cada fonte, por ordem de identificador.
159        """
160        return {source_id: self.score(source_id) for source_id in sorted(set(source_ids))}
161
162    def update(
163        self, observation: Observation, incident_id: str, result: EvaluationResult,
164        *, run_id: str | None = None,
165    ) -> dict:
166        """Aplicar um resultado à reputação da fonte e devolver o registo de auditoria.
167
168        As contagens só mudam se ``rejection_reason`` for ``None`` e nem o evento
169        nem o par fonte/incidente tiverem já sido avaliados neste ``run_id``. O
170        registo é sempre acrescentado a ``history``.
171
172        Args:
173            observation (Observation): Observação cuja fonte é avaliada.
174            incident_id (str): Incidente a que a observação pertence.
175            result (EvaluationResult): Adjudicação a aplicar.
176            run_id (str | None): Execução de avaliação; ``None`` partilha um único âmbito.
177
178        Returns:
179            dict: Registo de auditoria; ``updated`` e ``reason`` dizem se houve alteração
180                e porquê, e ``before``/``after`` dão o score antes e depois.
181
182        Raises:
183            TypeError: Se ``observation`` não for ``Observation`` ou ``result`` não for
184                ``EvaluationResult``.
185            ValueError: Se ``run_id`` for dado mas estiver vazio.
186        """
187        if not isinstance(observation, Observation) or not isinstance(result, EvaluationResult):
188            raise TypeError("São necessários Observation e EvaluationResult.")
189        if run_id is not None and (not isinstance(run_id, str) or not run_id.strip()):
190            raise ValueError("run_id deve identificar a execução de avaliação.")
191        reason = result.rejection_reason(observation, incident_id)
192        event_key = (run_id, observation.event_id)
193        key = (run_id, observation.source_id, incident_id)
194        if reason is None and (
195            event_key in self._evaluated_events or key in self._evaluated_incidents
196        ):
197            reason = "already_evaluated"
198        before = self.score(observation.source_id)
199        if reason is None:
200            correct, incorrect = self._counts.get(observation.source_id, (0, 0))
201            self._counts[observation.source_id] = (
202                correct + (result.outcome == "confirmed"),
203                incorrect + (result.outcome == "refuted"),
204            )
205            self._evaluated_events.add(event_key)
206            self._evaluated_incidents.add(key)
207        record = {
208            "event_id": observation.event_id, "source_id": observation.source_id,
209            "incident_id": incident_id, "run_id": run_id, "result": asdict(result),
210            "updated": reason is None, "reason": reason or "grounded_result",
211            "before": before, "after": self.score(observation.source_id),
212            "counts": self._counts.get(observation.source_id, (0, 0)),
213            "alpha": self.alpha, "beta": self.beta,
214        }
215        self.history.append(record)
216        return record

219def result_from_mission(
220    result, observation: Observation, incident_id: str,
221    *, max_distance: float, max_delay_seconds: float,
222) -> EvaluationResult:
223    """Adjudicar automaticamente um ``MISSION_RESULT`` já validado pelo sensor.
224
225    Só ``verify_incident`` confirmado é final. ``not_confirmed`` não exclui a
226    ameaça e fica ``inconclusive``. Há cobertura quando a tarefa é
227    ``verify_incident`` e a célula inspecionada e a conclusão da inspeção (não
228    anterior à observação) estão dentro dos limites experimentais dados.
229
230    Args:
231        result (Message): ``MISSION_RESULT`` recebido do recurso móvel.
232        observation (Observation): Observação a adjudicar.
233        incident_id (str): Incidente a que a observação pertence.
234        max_distance (float): Distância Manhattan máxima, em células, entre a célula
235            inspecionada e a observação.
236        max_delay_seconds (float): Atraso máximo, em segundos simulados, entre a
237            observação e a conclusão da inspeção.
238
239    Returns:
240        EvaluationResult: Adjudicação ``confirmed`` (final) ou ``inconclusive``, com
241            fundamento ``independent_mission``.
242
243    Raises:
244        ValueError: Se um limite de cobertura for negativo ou não finito.
245    """
246    for value in (max_distance, max_delay_seconds):
247        if not is_finite_number(value) or value < 0:
248            raise ValueError("Limites de cobertura devem ser finitos e não negativos.")
249    payload = result.payload
250    try:
251        position = Position(**payload["position"])
252        gap = abs(position.x - observation.position.x) + abs(position.y - observation.position.y)
253        delay = payload["completed_at"] - observation.observed_at
254        covered = (payload.get("task") == "verify_incident"
255                   and gap <= max_distance and 0 <= delay <= max_delay_seconds)
256    except (KeyError, TypeError, ValueError):
257        covered = False  # Posição inválida/ausente não comprova cobertura.
258    confirmed = payload.get("outcome") == "confirmed" and payload.get("task") == "verify_incident"
259    return EvaluationResult(
260        result_id=f"{result.message_id}:evaluation:{observation.event_id}",
261        incident_id=incident_id, observation_ids=(observation.event_id,),
262        outcome="confirmed" if confirmed else "inconclusive", basis="independent_mission",
263        evaluated_at=result.created_at, evidence_ids=(result.message_id,),
264        evidence_source_ids=(result.source_id,), final=confirmed, sufficient_coverage=covered,
265    )

224    """Adjudicar automaticamente um ``MISSION_RESULT`` já validado pelo sensor.
225
226    Só ``verify_incident`` confirmado é final. ``not_confirmed`` não exclui a

246    """

251    try:

264        evaluated_at=result.created_at, evidence_ids=(result.message_id,),
