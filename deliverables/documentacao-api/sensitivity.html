 77def alternative_matches(value, candidate):
 78    """Dizer se um valor alternativo corresponde ao pedido.
 79
 80    Números comparam-se por valor (2 e 2.0 equivalem; NaN e infinitos nunca); listas, elemento
 81    a elemento. JSON booleano/null nunca equivale a um número.
 82
 83    Args:
 84        value (object): Valor pedido ou lido (ex.: de ``--value`` ou de um relatório).
 85        candidate (object): Alternativa prevista em ``PARAMETERS``.
 86
 87    Returns:
 88        bool: ``True`` se os dois representarem a mesma alternativa.
 89    """
 90    if type(value) in (int, float) and type(candidate) in (int, float):
 91        return math.isfinite(value) and value == candidate
 92    if isinstance(value, list) and isinstance(candidate, list):
 93        return len(value) == len(candidate) and all(
 94            alternative_matches(left, right) for left, right in zip(value, candidate))
 95    return type(value) is type(candidate) and value == candidate

 98def apply_experiment(config, parameter, value):
 99    """Devolver cópias independentes (cenário, opções do runner) de um só fator.
100
101    Args:
102        config (dict): Cenário original, que não é alterado.
103        parameter (str): Fator experimental (chave de ``PARAMETERS``).
104        value (object): Alternativa a aplicar; tem de ser uma das previstas.
105
106    Returns:
107        tuple[dict, dict]: O cenário com o fator aplicado e as opções extra do runner.
108
109    Raises:
110        ValueError: Se o parâmetro for desconhecido, a alternativa não estiver prevista ou a
111            configuração resultante for inválida (ver ``validate_configuration``).
112        BlockedExperiment: Para ``source_diversity=1``, que exige aprovação explícita.
113    """
114    if parameter not in PARAMETERS:
115        raise ValueError(f"Parâmetro desconhecido: {parameter}.")
116    spec = PARAMETERS[parameter]
117    if not any(alternative_matches(value, candidate) for candidate in spec["alternatives"]):
118        raise ValueError(f"{parameter}: alternativa inválida {value!r}.")
119    if parameter == "source_diversity" and value == 1:
120        raise BlockedExperiment(BLOCKED_DIVERSITY)
121    result, options = deepcopy(config), {}
122    settings = result["response_settings"]
123    if parameter in ("corroboration_distance", "corroboration_window"):
124        key = "max_distance" if parameter.endswith("distance") else "max_time_difference"
125        options["corroboration_settings"] = {key: value}
126    elif parameter == "receive_timeout":
127        options["receive_timeout"] = value
128    elif parameter == "minimum_autonomy":
129        options["minimum_autonomy_percent"] = value
130    elif parameter == "cost_weights":
131        options["selection_weights"] = dict(zip(
132            ("eta_weight", "energy_weight", "load_weight"), value))
133    elif parameter == "sensor_quality":
134        rate = 0.0 if value == "ideal" else 0.3
135        for source in result["sources"].values():
136            source.update(false_positive_rate=rate, false_negative_rate=rate)
137    elif parameter == "reputation_prior":
138        settings["reputation"].update(alpha=value, beta=value)
139    elif parameter in ("patrol_threshold", "ranger_threshold"):
140        settings["policy"][parameter] = value
141    elif parameter == "source_diversity":
142        settings["policy"].update(ranger_min_groups=value, ranger_min_modalities=value)
143    elif parameter == "severity":
144        settings["severity"] = value
145    elif parameter == "adjudication_distance":
146        settings["adjudication"]["max_distance"] = value
147    elif parameter == "adjudication_delay":
148        settings["adjudication"]["max_delay_seconds"] = value
149    validate_configuration(result, options)
150    return result, options

179def compare_rows(actual, expected, *, identity):
180    """Comparar todos os indicadores comuns; N/A e arredondamentos são explícitos.
181
182    A referência (``expected``) define os campos comparados; campos extra em ``actual`` são
183    ignorados. Números comparam-se com tolerância de 1e-10.
184
185    Args:
186        actual (list[dict]): Linhas obtidas.
187        expected (list[dict]): Linhas de referência.
188        identity (tuple[str, ...]): Campos que identificam cada linha nos dois lados.
189
190    Returns:
191        list[dict]: Uma diferença por campo divergente ou linha em falta, com ``identity``,
192            ``field``, ``actual`` e ``expected``; vazia se coincidirem.
193
194    Raises:
195        ValueError: Se uma identidade se repetir em qualquer dos lados.
196    """
197    def key(row):
198        """Devolver a identidade da linha: os campos de ``identity`` em texto."""
199        return tuple(str(row[name]) for name in identity)
200
201    def same(left, right):
202        """Comparar dois valores: N/A só com N/A, números por ``isclose``, o resto em texto."""
203        if left is None or right is None or left == "N/A" or right == "N/A":
204            return left in (None, "N/A") and right in (None, "N/A")
205        if isinstance(left, bool) or right in ("True", "False"):
206            return str(left) == str(right)
207        try:
208            return math.isclose(float(left), float(right), rel_tol=1e-10, abs_tol=1e-10)
209        except (TypeError, ValueError):
210            return str(left) == str(right)
211
212    actual_map, expected_map = {}, {}
213    for rows, indexed in ((actual, actual_map), (expected, expected_map)):
214        for row in rows:
215            ident = key(row)
216            if ident in indexed:
217                raise ValueError(f"Linha duplicada na comparação: {ident}.")
218            indexed[ident] = row
219    differences = []
220    for ident in sorted(actual_map.keys() | expected_map.keys()):
221        left, right = actual_map.get(ident), expected_map.get(ident)
222        if left is None or right is None:
223            differences.append({"identity": ident, "field": "row", "actual": left, "expected": right})
224            continue
225        # A referência define o contrato de indicadores; extras experimentais não contam.
226        for field, expected_value in right.items():
227            if field not in left or not same(left[field], expected_value):
228                differences.append({"identity": ident, "field": field,
229                                    "actual": left.get(field), "expected": expected_value})
230    return differences

260def write_csv(path, rows):
261    """Gravar as linhas num CSV novo, com ``None`` como "N/A".
262
263    As colunas são a união dos campos, pela ordem em que aparecem. Sem linhas não cria
264    ficheiro; falha se o ficheiro já existir.
265
266    Args:
267        path (Path): Ficheiro a criar.
268        rows (list[dict]): Linhas a gravar.
269    """
270    if not rows:
271        return
272    fields = list(dict.fromkeys(field for row in rows for field in row))
273    with path.open("x", encoding="utf-8", newline="") as stream:
274        writer = csv.DictWriter(stream, fields, lineterminator="\n")
275        writer.writeheader()
276        writer.writerows({key: "N/A" if value is None else value for key, value in row.items()}
277                         for row in rows)

280def create_output(run_id=None):
281    """Criar a pasta de saída da recolha.
282
283    A raiz de escrita é fixa e o mkdir exclusivo impede substituir recolhas.
284
285    Args:
286        run_id (str | None): Nome da pasta em ``results/sensitivity/``; ``None`` gera um a partir
287            da data UTC e de um sufixo aleatório.
288
289    Returns:
290        Path: A pasta criada.
291
292    Raises:
293        ValueError: Se ``run_id`` não for um nome simples ou se a raiz de resultados for um
294            atalho para outro diretório.
295    """
296    name = run_id or datetime.now(timezone.utc).strftime("%Y%m%dT%H%M%SZ-") + uuid4().hex[:8]
297    if not re.fullmatch(r"[A-Za-z0-9][A-Za-z0-9._-]*", name) or name in (".", ".."):
298        raise ValueError("--run-id deve ser um nome simples, sem diretórios.")
299    if OUTPUT_ROOT.resolve() != OUTPUT_ROOT.absolute():
300        raise ValueError("A raiz de resultados não pode redirecionar para outro diretório.")
301    OUTPUT_ROOT.mkdir(parents=True, exist_ok=True)
302    output = OUTPUT_ROOT / name
303    output.mkdir(exist_ok=False)
304    return output

307def provenance():
308    """Descrever o código e o ambiente usados na recolha.
309
310    Inclui commit, ramo, alterações sem commit, versões de Python e SPADE, plataforma e
311    SHA-256 das fontes, cenários, ``requirements.txt`` e ficheiros de referência.
312
313    Returns:
314        dict: ``commit``, ``branch``, ``dirty``, ``git_status``, ``python``, ``platform``,
315            ``spade``, ``sha256`` (por ficheiro: fontes ``src``, este script,
316            ``config/**/*.json`` e ``requirements.txt``) e ``reference_sha256``.
317    """
318    def git(*args):
319        """Correr ``git`` na raiz do repositório e devolver o stdout aparado."""
320        return subprocess.run(["git", *args], cwd=ROOT, check=True,
321                              capture_output=True, text=True).stdout.strip()
322
323    files = sorted({*ROOT.glob("src/**/*.py"), ROOT / "scripts/sensitivity.py",
324                    *ROOT.glob("config/**/*.json"), ROOT / "requirements.txt"})
325    hashes = {str(path.relative_to(ROOT)): hashlib.sha256(path.read_bytes()).hexdigest()
326              for path in files}
327    status = git("status", "--porcelain", "--", "src", "config", "scripts", "tests")
328    return {"commit": git("rev-parse", "HEAD"), "branch": git("branch", "--show-current"),
329            "dirty": bool(status), "git_status": status,
330            "python": sys.version, "platform": platform.platform(),
331            "spade": importlib.metadata.version("spade"), "sha256": hashes,
332            "reference_sha256": {name: hashlib.sha256((REFERENCE_ROOT / name).read_bytes()).hexdigest()
333                                 for name in ("runs.csv", "summary.csv", "manifest.json")}}

336async def self_check(mode, configs, output):
337    """Comparar a referência por execução e, no modo completo, por agregado.
338
339    No modo ``initial`` corre só a primeira repetição do cenário A e compara-a com
340    ``runs.csv``; no ``full`` corre os cenários todos e compara execuções, ordem e resumo.
341    Grava ``actual.csv`` (e ``actual-runs.csv``) e ``comparison.json`` em ``output``.
342
343    Args:
344        mode (str): ``"initial"`` ou ``"full"``.
345        configs (list[dict]): Cenários a correr; ignorado no modo ``initial``.
346        output (Path): Pasta do self-check.
347
348    Returns:
349        dict: Resultado com ``passed``, ``run_count``, ``differences`` e, no modo ``full``, as
350            comparações por execução e por resumo.
351
352    Raises:
353        ValueError: Se a referência gravada não tiver as sementes, episódios ou ordem da Etapa 9.
354    """
355    actual, expected, run_count = [], [], 0
356    actual_runs, expected_runs, order_differences = [], [], []
357    run_identity = ("scenario", "variant", "repetition", "episode", "seed")
358    if mode == "initial":
359        configs = [load_scenario(SCENARIO_ROOT / "A.json")]
360        reference = read_csv(REFERENCE_ROOT / "runs.csv")
361        # A primeira repetição começa no prior original e mantém a ordem dos episódios.
362        expected = [row for row in reference if row["scenario"] == "A"
363                    and row["variant"] == "full" and row["repetition"] == "0"]
364        for row in expected:
365            if int(row["seed"]) != configs[0]["base_seed"]:
366                raise ValueError("Semente da referência inicial diverge do cenário A.")
367        identity = run_identity
368    else:
369        selected = {config["scenario_id"] for config in configs}
370        expected = [row for row in read_csv(REFERENCE_ROOT / "summary.csv")
371                    if row["variant"] == "full" and row["scenario"] in selected]
372        identity = ("scenario", "variant")
373        reference_runs = read_csv(REFERENCE_ROOT / "runs.csv")
374        for config in configs:
375            stored = [row for row in reference_runs if row["scenario"] == config["scenario_id"]
376                      and row["variant"] == "full"]
377            planned = [(str(rep), episode["label"], str(config["base_seed"] + rep))
378                       for rep in range(config["repetitions"]) for episode in config["episodes"]]
379            if [(row["repetition"], row["episode"], row["seed"]) for row in stored] != planned:
380                raise ValueError("A referência completa exige a mesma ordem, episódios e sementes da Etapa 9.")
381            expected_runs.extend(stored)
382    for config in configs:
383        rows = await run_scenario(config, "full", repetitions=1 if mode == "initial" else None,
384                                  output=output / "runs")
385        run_count += len(rows)
386        actual.extend(rows if mode == "initial" else [flat_summary(config["scenario_id"], "full", rows)])
387        if mode == "full":
388            actual_runs.extend(rows)
389            stored = [row for row in expected_runs if row["scenario"] == config["scenario_id"]]
390            received_order = [tuple(str(row[name]) for name in run_identity) for row in rows]
391            expected_order = [tuple(row[name] for name in run_identity) for row in stored]
392            if received_order != expected_order:
393                order_differences.append({
394                    "identity": [config["scenario_id"], "full"], "field": "execution_order",
395                    "actual": received_order, "expected": expected_order,
396                })
397    differences = compare_rows(actual, expected, identity=identity)
398    result = {"mode": mode, "passed": not differences, "compared_rows": len(actual),
399              "run_count": run_count,
400              "differences": differences, "reference": "runs.csv" if mode == "initial" else "summary.csv"}
401    if mode == "full":
402        run_differences = compare_rows(actual_runs, expected_runs, identity=run_identity)
403        run_differences.extend(order_differences)
404        result["run_comparison"] = {
405            "passed": not run_differences, "compared_rows": len(actual_runs),
406            "differences": run_differences, "reference": "runs.csv",
407        }
408        result["summary_comparison"] = {
409            "passed": not differences, "compared_rows": len(actual),
410            "differences": differences, "reference": "summary.csv",
411        }
412        result["passed"] = not run_differences and not differences
413        result["differences"] = [
414            {"level": level, **difference}
415            for level, items in (("runs", run_differences), ("summary", differences))
416            for difference in items
417        ]
418        result["references"] = ["runs.csv", "summary.csv"]
419        write_csv(output / "actual-runs.csv", actual_runs)
420    write_csv(output / "actual.csv", actual)
421    write_json(output / "comparison.json", result)
422    return result

425def plan_experiments(configs, parameters, value=UNSET, repetitions=None):
426    """Construir o plano: um item por parâmetro, alternativa e cenário.
427
428    Cada item traz a configuração aplicada, ou o estado ``blocked``/``invalid`` com o
429    motivo, e as repetições, sementes e episódios previstos.
430
431    Args:
432        configs (list[dict]): Cenários originais.
433        parameters (list[str]): Fatores a incluir.
434        value (object): Restringe o plano a uma alternativa; ``UNSET`` inclui todas.
435        repetitions (int | None): Repetições por cenário; ``None`` usa as do cenário.
436
437    Returns:
438        list[dict]: Itens com ``parametro``, ``valor``, ``cenario``, ``status``, ``repetitions``,
439            ``seeds``, ``episodes``, ``runs`` e ainda ``config`` e ``runner_options`` (se
440            ``ready``) ou ``reason`` (se não).
441
442    Raises:
443        ValueError: Se nenhuma alternativa corresponder a ``value``.
444    """
445    plan = []
446    for parameter in parameters:
447        for alternative in PARAMETERS[parameter]["alternatives"]:
448            if value is not UNSET and not alternative_matches(value, alternative):
449                continue
450            for original in configs:
451                item = {"parametro": parameter, "valor": alternative,
452                        "cenario": original["scenario_id"], "status": "ready"}
453                try:
454                    config, options = apply_experiment(original, parameter, alternative)
455                    item.update(config=config, runner_options=options)
456                except BlockedExperiment as error:
457                    item.update(status="blocked", reason=str(error))
458                except (TypeError, ValueError, KeyError) as error:
459                    item.update(status="invalid", reason=str(error))
460                count = repetitions if repetitions is not None else original["repetitions"]
461                item.update(repetitions=count, seeds=list(range(original["base_seed"], original["base_seed"] + count)),
462                            episodes=[episode["label"] for episode in original["episodes"]],
463                            runs=count * len(original["episodes"]))
464                plan.append(item)
465    if not plan:
466        raise ValueError("Nenhuma alternativa corresponde a --value.")
467    return plan

470def report_extras(records):
471    """Somar os indicadores extra de um grupo de relatórios sobre as execuções.
472
473    Execuções expiradas, contagens por motivo final, recusas por autonomia, missões e
474    CFP de rangers e atualizações de reputação.
475
476    Args:
477        records (list[dict]): Ficheiros de relatório, com a chave ``report``.
478
479    Returns:
480        dict: ``expired_runs``, ``reason_<motivo>`` para quatro motivos,
481            ``autonomy_insufficient``, ``ranger_missions_known``, ``ranger_cfp_sent`` e
482            ``reputation_updates``.
483    """
484    reports = [record["report"] for record in records]
485    reasons = Counter(report["reason"] for report in reports)
486    return {"expired_runs": sum(report["incident_status"] == "expired" for report in reports),
487            **{f"reason_{reason}": reasons[reason] for reason in (
488                "insufficient_corroboration", "no_eligible_proposals", "confirmation_unknown", "mission_result_unknown")},
489            "autonomy_insufficient": sum(entry["body"]["payload"].get("reason") == "autonomy_insufficient"
490                                         for report in reports for entry in report["messages"]),
491            "ranger_missions_known": sum(mission["task"] == "respond_incident"
492                                    for report in reports for mission in report["missions"]),
493            "ranger_cfp_sent": sum(entry["body"].get("message_type") == "CFP"
494                                   and entry["body"]["payload"].get("task") == "respond_incident"
495                                   for report in reports for entry in report["messages"]),
496            "reputation_updates": sum(update.get("updated", False)
497                                       for report in reports for update in report["reputation_updates"])}
