  • isia.agents.incident: IncidentSensorAgent, o sensor que inicia e coordena uma ocorrência.
  • isia.agents.sensor: SensorAgent, a base do iniciador; pede e avalia a verificação.
  • isia.agents.informant: InformantAgent, responde a pedidos de verificação com as suas observações.
  • isia.agents.mobile: MobileResourceAgent, responde a CFP, reserva e executa missões.

153    def configure_execution(self, world, inspect, *, speed=1.0, energy_per_cell=1.0,
154                            service_seconds=1.0):
155        """Ativar a execução simulada de missões em ``world``.
156
157        A posição do recurso é validada no mundo.
158
159        Args:
160            world (World): Mapa onde o recurso se move.
161            inspect (Callable[[Position, float, str], str]): Inspeção local no alvo; recebe a
162                posição, o instante simulado e a tarefa e devolve o resultado.
163            speed (float): Velocidade, em células por segundo simulado; tem de ser positiva.
164            energy_per_cell (float): Energia gasta, em %, por célula percorrida.
165            service_seconds (float): Duração da inspeção no alvo, em segundos simulados.
166
167        Raises:
168            ValueError: Se algum parâmetro for negativo, a velocidade for nula, ``inspect`` não
169                for invocável ou a posição do recurso estiver fora do mapa.
170        """
171        for name, value in (("speed", speed), ("energy_per_cell", energy_per_cell),
172                            ("service_seconds", service_seconds)):
173            nonnegative(value, name)
174        if speed == 0 or not callable(inspect):
175            raise ValueError("Velocidade positiva e inspetor são obrigatórios.")
176        world.validate_position(self.resource.position)
177        self.execution_options = dict(world=world, inspect=inspect, speed=speed,
178                                      energy_per_cell=energy_per_cell, service_seconds=service_seconds,
179                                      minimum_energy=self.minimum_autonomy_percent)

181    def is_turn(self, requester, priority):
182        """Registar quem pede na fila ordenada e dizer se é a sua vez.
183
184        Primeiro a ocorrência mais grave; em caso de empate, a que espera há mais tempo.
185        Pedidos não renovados dentro de ``WAITING_TTL_SECONDS`` simulados ou
186        ``WAITING_TTL_REAL_SECONDS`` reais saem da fila.
187
188        Args:
189            requester (tuple[str, str]): ``(JID do iniciador, mission_id)`` do pedido.
190            priority (int): Prioridade da missão; maior é mais grave.
191
192        Returns:
193            bool: True se este pedido estiver à frente da fila.
194        """
195        now, real = self.simulated_time, monotonic()
196        self._waiting = {key: entry for key, entry in self._waiting.items()
197                         if now - entry[3] <= WAITING_TTL_SECONDS and real - entry[4] <= WAITING_TTL_REAL_SECONDS}
198        since = self._waiting.get(requester, (0, now, real))[1:3]
199        self._waiting[requester] = (-priority, *since, now, real)
200        # A ordem só usa o que não muda entre pedidos; o último pedido serve apenas para a validade.
201        return min(self._waiting, key=lambda key: self._waiting[key][:3]) == requester

203    def returning(self):
204        """Encadear missões: a caminho da base e com o resultado entregue, o recurso já pode ser pedido.
205
206        Returns:
207            bool: True com ``chain_missions`` ativo, o resultado já enviado e o recurso em
208                ``RETURNING``.
209        """
210        execution = self.execution
211        return (self.chain_missions and execution is not None and not execution.finished
212                and self._result_sent and execution.resource.status is ResourceStatus.RETURNING)

214    def _close_execution(self):
215        """Fechar a proposta da execução atual: deixa de contar como aceite."""
216        accept, _ = self._execution_context
217        proposal_id = accept.payload["proposal_message_id"]
218        self._accepted_proposal_ids.discard(proposal_id)
219        self._invalidate_proposal(proposal_id)

221    def evaluate_request(self, cfp):
222        """Avaliar um CFP e devolver PROPOSAL ou REJECT, sem alterar a reserva.
223
224        Com execução configurada, o ETA vem da distância ao alvo e a energia da
225        ida e do regresso, e um alvo fora do mapa é recusado com ``target_outside_map``.
226        Ocupado numa missão por terminar, a recusa leva ``busy_until``.
227
228        Args:
229            cfp (Message): CFP recebido do iniciador.
230
231        Returns:
232            Message: ``PROPOSAL`` com ETA, custo e carga, ou ``REJECT`` com o motivo.
233        """
234        eta, energy = self.eta_seconds, self.energy_cost_percent
235        state = self.reservation.resource_state
236        if self.returning():
237            state = state.change_status(ResourceStatus.AVAILABLE)  # só para avaliar; a reserva não muda
238        if self.execution_options is not None:
239            target = Position(**cfp.payload["target"])
240            if not self.execution_options["world"].contains(target):
241                return Message(
242                    f"reject-{cfp.message_id}-{self.resource.resource_id}", MessageType.REJECT,
243                    self.simulated_time, {"cfp_message_id": cfp.message_id, "reason": "target_outside_map"},
244                    mission_id=cfp.mission_id, incident_id=cfp.incident_id,
245                    conversation_id=cfp.conversation_id, source_id=self.resource.resource_id)
246            cells = distance(state.position, target)
247            eta = cells / self.execution_options["speed"]
248            # Ida a partir da posição atual e regresso à base; parado na base, é o dobro da ida.
249            home = self.base if self.chain_missions else state.position
250            energy = (cells + distance(target, home)) * self.execution_options["energy_per_cell"]
251        response = evaluate_cfp(
252            cfp, state, self.capabilities,
253            current_time=self.simulated_time, minimum_autonomy_percent=self.minimum_autonomy_percent,
254            eta_seconds=eta, energy_cost_percent=energy, load=self.load)
255        if (response.payload.get("reason") == "resource_unavailable"
256                and self.execution is not None and not self.execution.finished):
257            # Ocupado numa missão: dizer quando fica livre, para o iniciador não perguntar de 5 em 5 s.
258            response = Message(
259                response.message_id, MessageType.REJECT, response.created_at,
260                {**response.payload, "busy_until": self.execution.free_at(self.chain_missions)},
261                mission_id=cfp.mission_id, incident_id=cfp.incident_id,
262                conversation_id=cfp.conversation_id, source_id=self.resource.resource_id)
263        return response

265    def start_execution(self, cfp, accept, initiator):
266        """Iniciar a execução da missão aceite, uma única vez por ACCEPT.
267
268        Não faz nada sem execução configurada, com outra missão ainda em curso ou
269        se este ACCEPT já tiver sido executado.
270
271        Args:
272            cfp (Message): CFP da missão; dá a tarefa, o alvo, a capacidade e a prioridade.
273            accept (Message): ACCEPT que confirmou a reserva.
274            initiator (str): JID do iniciador, destinatário do MISSION_RESULT.
275        """
276        if self.execution_options is None or accept.message_id in self._completed_accepts:
277            return
278        if self.execution is not None and not self.execution.finished:
279            return
280        mission = Mission(cfp.mission_id, cfp.incident_id, cfp.payload["task"],
281                          Position(**cfp.payload["target"]), cfp.payload["required_capability"],
282                          cfp.payload["priority"])
283        self.execution = MissionExecution(self.reservation, mission, now=self.simulated_time,
284                                          home=self.base if self.chain_missions else None,
285                                          **self.execution_options)
286        self.executions.append(self.execution)
287        self._execution_context = (accept, initiator)
288        self._completed_accepts.add(accept.message_id)
289        self._result_sent = False
290        self._result_message = None

292    class ExecutionBehaviour(CyclicBehaviour):
293        """Avançar a missão com o relógio simulado e enviar o MISSION_RESULT."""
294        async def run(self):
295            """Avançar a execução até ao instante simulado e enviar o resultado uma vez.
296
297            Se o envio falhar, a mesma mensagem é tentada de novo no ciclo seguinte.
298
299            Overrides:
300                spade.behaviour.CyclicBehaviour.run
301            """
302            agent = self.agent
303            await asyncio.sleep(0.001)
304            execution = agent.execution
305            if execution is None:
306                return
307            execution.advance(agent.simulated_time)
308            agent.resource = execution.resource
309            if execution.finished:
310                agent._close_execution()
311            if execution.result is not None and not agent._result_sent:
312                accept, initiator = agent._execution_context
313                if agent._result_message is None:
314                    agent._result_message = Message(
315                        f"result-{accept.message_id}", MessageType.MISSION_RESULT,
316                        agent.simulated_time,
317                        {**execution.result, "accept_message_id": accept.message_id,
318                         "resource_id": agent.resource.resource_id, "task": execution.mission.task},
319                        incident_id=accept.incident_id, mission_id=accept.mission_id,
320                        source_id=agent.resource.resource_id, conversation_id=accept.conversation_id)
321                result = agent._result_message
322                outgoing = build_spade_message(
323                    initiator, result, performative=PERFORMATIVE_INFORM,
324                    conversation_id=result.conversation_id, in_reply_to=accept.message_id,
325                    protocol=CONTRACT_NET_PROTOCOL)
326                try:
327                    await self.send(outgoing)
328                except Exception as error:
329                    logger.warning("Resultado não enviado; o regresso continua: %s", error)
330                    return
331                agent._result_sent = True
332                if accept.message_id in agent._accept_replies:
333                    agent._accept_replies[accept.message_id].append(outgoing)

335    def set_simulated_time(self, simulated_time: float) -> None:
336        """Avançar o relógio simulado e expirar propostas; ValueError se recuar.
337
338        Args:
339            simulated_time (float): Novo instante simulado, em segundos.
340
341        Raises:
342            ValueError: Se o instante não for finito e não negativo, ou for anterior ao atual.
343        """
344        value = _validate_simulated_time(simulated_time, "simulated_time")
345        if value < self.simulated_time:
346            raise ValueError("O relógio do recurso não pode recuar.")
347        self.simulated_time = value
348        self._expire_proposals()

350    def _invalidate_proposal(self, proposal_id: str) -> None:
351        """Esquecer a proposta e fechar o seu CFP, que deixa de ser reavaliado."""
352        context = self._proposal_context.pop(proposal_id, None)
353        if context is not None:
354            self._closed_cfp_ids.add(context[0].message_id)

356    def _expire_proposals(self) -> None:
357        """Invalidar as propostas não aceites cujo prazo de aceitação já passou.
358
359        O prazo é o ``deadline`` do CFP mais ``PROPOSAL_ACCEPTANCE_WINDOW_SECONDS``.
360
361        Raises:
362            ValueError: Se o relógio simulado do agente não for finito e não negativo.
363        """
364        _validate_simulated_time(self.simulated_time, "simulated_time")
365        for proposal_id, (cfp, _, _) in tuple(self._proposal_context.items()):
366            if proposal_id in self._accepted_proposal_ids:
367                continue  # Preservar confirmações idempotentes de reservas existentes.
368            if (
369                self.simulated_time - cfp.payload["deadline"]
370                > PROPOSAL_ACCEPTANCE_WINDOW_SECONDS
371            ):
372                self._invalidate_proposal(proposal_id)

374    class ContractNetBehaviour(CyclicBehaviour):
375        """Responder a cada CFP com uma proposta ou uma recusa."""
376        async def run(self):
377            """Responder a um CFP com PROPOSAL ou REJECT, de forma idempotente.
378
379            Um CFP repetido recebe a mesma proposta; CFP fechados, mais antigos ou de
380            rondas anteriores são ignorados; com ``resend_replies``, a recusa já dada
381            a esse CFP é repetida. Uma ronda mais recente invalida as
382            propostas não aceites da mesma negociação.
383
384            Overrides:
385                spade.behaviour.CyclicBehaviour.run
386            """
387            incoming = await self.receive(timeout=10)
388            if incoming is None:
389                return
390            try:
391                cfp = parse_spade_message(
392                    incoming,
393                    expected_protocol=CONTRACT_NET_PROTOCOL,
394                )
395                if cfp.message_type is not MessageType.CFP:
396                    return
397                agent: MobileResourceAgent = self.agent  # type: ignore[assignment]
398                agent._expire_proposals()
399                initiator = _bare_jid(incoming.sender)
400                negotiation = (initiator, cfp.mission_id)
401                if (
402                    cfp.message_id in agent._closed_cfp_ids
403                    or cfp.created_at < agent._latest_cfp_times.get(negotiation, -1)
404                ):
405                    refusal = agent._refusals.get((initiator, cfp.message_id))
406                    if refusal is not None:
407                        await self.send(refusal)  # a recusa perdeu-se: repetir a mesma, sem reavaliar
408                    return
409                existing = next((
410                    context for context in agent._proposal_context.values()
411                    if context[0].message_id == cfp.message_id
412                    and _bare_jid(context[2]) == initiator
413                ), None)
414                if existing is not None and existing[0].to_dict() != cfp.to_dict():
415                    logger.warning("CFP repetido com conteúdo divergente: %s", cfp.message_id)
416                    return
417                round_number = cfp.payload.get("round_number")
418                latest_round = agent._latest_cfp_rounds.get(negotiation)
419                if round_number is not None and latest_round is not None:
420                    latest_number, latest_cfp_id = latest_round
421                    if round_number < latest_number or (
422                        round_number == latest_number and cfp.message_id != latest_cfp_id
423                    ):
424                        return
425                response = existing[1] if existing is not None else agent.evaluate_request(cfp)
426                if agent.ordered_queue and existing is None and (
427                        response.message_type is MessageType.PROPOSAL
428                        or response.payload.get("reason") == "resource_unavailable"):
429                    # Ocupado ou livre, o pedido entra na fila; livre, só o primeiro recebe proposta.
430                    if not agent.is_turn(negotiation, cfp.payload["priority"]) and (
431                            response.message_type is MessageType.PROPOSAL):
432                        response = Message(
433                            f"reject-{cfp.message_id}-{agent.resource.resource_id}", MessageType.REJECT,
434                            agent.simulated_time, {"cfp_message_id": cfp.message_id, "reason": "waiting_turn"},
435                            mission_id=cfp.mission_id, incident_id=cfp.incident_id,
436                            conversation_id=cfp.conversation_id, source_id=agent.resource.resource_id)
437                agent._latest_cfp_times[negotiation] = cfp.created_at
438                if round_number is not None:
439                    agent._latest_cfp_rounds[negotiation] = (round_number, cfp.message_id)
440                for proposal_id, (previous_cfp, _, previous_sender) in tuple(agent._proposal_context.items()):
441                    previous_round = previous_cfp.payload.get("round_number")
442                    # Sem ordinais comparáveis, um empate temporal não prova
443                    # sucessão: conservar os contextos legados separadamente.
444                    newer_round = (
445                        round_number > previous_round
446                        if round_number is not None and previous_round is not None
447                        else cfp.created_at > previous_cfp.created_at
448                    )
449                    if (
450                        previous_cfp.mission_id == cfp.mission_id
451                        and _bare_jid(previous_sender) == initiator
452                        and previous_cfp.message_id != cfp.message_id
453                        and newer_round
454                        and proposal_id not in agent._accepted_proposal_ids
455                    ):
456                        agent._invalidate_proposal(proposal_id)
457                if response.message_type is MessageType.PROPOSAL:
458                    performative = PERFORMATIVE_PROPOSE
459                else:
460                    performative = PERFORMATIVE_REFUSE
461                reply = build_spade_message(
462                    str(incoming.sender),
463                    response,
464                    performative=performative,
465                    conversation_id=cfp.conversation_id,
466                    in_reply_to=cfp.message_id,
467                    protocol=CONTRACT_NET_PROTOCOL,
468                )
469                if response.message_type is MessageType.PROPOSAL:
470                    proposal_context = existing or (
471                        cfp,
472                        response,
473                        str(incoming.sender),
474                    )
475                    agent._proposal_context[response.message_id] = proposal_context
476                else:
477                    proposal_context = None
478                    agent._closed_cfp_ids.add(cfp.message_id)
479                    if agent.resend_replies:
480                        agent._refusals[(initiator, cfp.message_id)] = reply
481                try:
482                    await self.send(reply)
483                except Exception as error:
484                    if response.message_type is MessageType.PROPOSAL and existing is None:
485                        if agent._proposal_context.get(response.message_id) == proposal_context:
486                            agent._proposal_context.pop(response.message_id, None)
487                    logger.warning("Resposta Contract Net não enviada: %s", error)
488            except (TypeError, ValueError) as error:
489                logger.warning("CFP rejeitado: %s", error)

491    class AcceptBehaviour(CyclicBehaviour):
492        """Tratar o ACCEPT ou o REJECT do iniciador a uma proposta deste recurso."""
493        async def run(self):
494            """Reservar o recurso num ACCEPT válido e responder com um STATUS.
495
496            Um REJECT correlacionado só invalida a proposta. Um ACCEPT sem proposta
497            conhecida recebe sempre um STATUS de pedido inválido; um ACCEPT incoerente
498            recebe o mesmo STATUS, ou silêncio com ``silent_invalid``. Com
499            ``ordered_queue`` e outro pedido à frente, o STATUS é de recurso ocupado
500            (``waiting_turn``). Um ACCEPT já executado só repete as respostas guardadas.
501            Com a reserva confirmada, a execução arranca mesmo que a confirmação não
502            chegue a ser enviada.
503
504            Overrides:
505                spade.behaviour.CyclicBehaviour.run
506            """
507            incoming = await self.receive(timeout=10)
508            if incoming is None:
509                return
510            agent: MobileResourceAgent = self.agent  # type: ignore[assignment]
511            try:
512                accept = parse_spade_message(
513                    incoming,
514                    expected_protocol=CONTRACT_NET_PROTOCOL,
515                )
516                expected_performative = {
517                    MessageType.ACCEPT: PERFORMATIVE_ACCEPT_PROPOSAL,
518                    MessageType.REJECT: PERFORMATIVE_REJECT_PROPOSAL,
519                }.get(accept.message_type)
520                if expected_performative is None or (
521                    incoming.get_metadata("performative") != expected_performative
522                ):
523                    logger.warning("Decisão Contract Net rejeitada: intenção e corpo incoerentes")
524                    return
525                agent._expire_proposals()
526                context = agent._proposal_context.get(
527                    accept.payload.get("proposal_message_id")
528                )
529                if accept.message_type is MessageType.REJECT:
530                    if context is not None:
531                        cfp, proposal, initiator = context
532                        if (
533                            _bare_jid(incoming.sender) == _bare_jid(initiator)
534                            and _bare_jid(getattr(incoming, "to", "")) == _bare_jid(agent.jid)
535                            and (accept.source_id is None or _bare_jid(accept.source_id) == _bare_jid(initiator))
536                            and incoming.get_metadata("in-reply-to") == proposal.message_id
537                            and accept.mission_id == cfp.mission_id
538                            and accept.incident_id == cfp.incident_id
539                            and accept.conversation_id == cfp.conversation_id
540                            and accept.payload.get("cfp_message_id") == cfp.message_id
541                            and accept.payload.get("resource_id") == proposal.source_id
542                            and proposal.created_at <= accept.created_at <= agent.simulated_time
543                            and proposal.message_id not in agent._accepted_proposal_ids
544                        ):
545                            agent._invalidate_proposal(proposal.message_id)
546                            agent._waiting.pop((_bare_jid(initiator), cfp.mission_id), None)
547                    return
548                if accept.message_id in agent._completed_accepts:
549                    # Uma missão terminada nunca volta a reservar por replay; com reenvio, repete
550                    # a confirmação e o resultado já enviados ao iniciador original.
551                    for stored in tuple(agent._accept_replies.get(accept.message_id, ())):
552                        await self.send(stored)
553                    return
554                if context is None:
555                    result = ReservationResult(
556                        ReservationStatus.INVALID_REQUEST,
557                        agent.reservation.resource_state,
558                        accept.mission_id,
559                        "proposal_not_correlated",
560                    )
561                else:
562                    cfp, proposal, initiator = context
563                    valid_sender = _bare_jid(incoming.sender) == _bare_jid(initiator)
564                    valid_recipient = _bare_jid(getattr(incoming, "to", "")) == _bare_jid(agent.jid)
565                    valid_source = accept.source_id is None or _bare_jid(accept.source_id) == _bare_jid(initiator)
566                    if not valid_sender or not valid_source or not accept_matches_proposal(
567                        accept, cfp, proposal, agent.reservation.resource_state.resource_id
568                    ) or not valid_recipient or (
569                        incoming.get_metadata("in-reply-to") != proposal.message_id
570                        or accept.created_at > agent.simulated_time
571                    ):
572                        if agent.silent_invalid:
573                            # Contra-amplificação: responder daria tráfego grátis a quem envia lixo.
574                            logger.warning("ACCEPT incoerente ignorado: %s", accept.message_id)
575                            return
576                        result = ReservationResult(
577                            ReservationStatus.INVALID_REQUEST,
578                            agent.reservation.resource_state,
579                            accept.mission_id,
580                            "accept_correlation_invalid",
581                        )
582                    elif (agent.ordered_queue and proposal.message_id not in agent._accepted_proposal_ids
583                          and not agent.is_turn((_bare_jid(initiator), cfp.mission_id), cfp.payload["priority"])):
584                        # Entre a proposta e o ACCEPT pediu alguém mais grave: a reserva fica para esse.
585                        result = ReservationResult(
586                            ReservationStatus.RESOURCE_OCCUPIED, agent.reservation.resource_state,
587                            accept.mission_id, "waiting_turn")
588                    else:
589                        if agent.returning():
590                            # Encadear: o regresso acaba aqui e a missão seguinte parte da posição atual.
591                            agent.execution.advance(agent.simulated_time)
592                            agent.execution.release()
593                            agent._close_execution()
594                        result = agent.reservation.reserve(
595                            accept.mission_id,
596                            resource_id=agent.reservation.resource_state.resource_id,
597                        )
598                        agent.resource = result.resource_state
599                        if result.confirmed:
600                            agent._accepted_proposal_ids.add(proposal.message_id)
601                            agent._waiting.pop((_bare_jid(initiator), cfp.mission_id), None)

222
223            Respostas inválidas ou mal correlacionadas são rejeitadas e a espera
224            continua. Sem resposta válida até ao prazo, fica a decisão de
225            ``_record_safe_observe``.
226
227            Overrides:
228                spade.behaviour.OneShotBehaviour.run
229            """
230            agent: SensorAgent = self.agent  # type: ignore[assignment]
231            conversation_id = f"conversation-{uuid.uuid4().hex}"
232            agent.conversation_id = conversation_id
233
234            # O iniciador de ocorrências já leu o feed: a verificação usa essa leitura e o mesmo incidente.
235            observations = getattr(agent, "incident_observations", None)
236            if not isinstance(observations, tuple) or not observations:
237                observations = _recent_observations(agent.observation_feed)
238            state = getattr(agent, "incident_state", None)
239
240            if observations:
241                observation = observations[0]
242                agent.initial_observation = observation
243                created_at = (
244                    agent.created_at
245                    if isinstance(agent.created_at, (int, float))
246                    else observation.observed_at
247                )
248                payload = {
249                    "observation": model_to_dict(observation),
250                    "hypothesis": build_observation_hypothesis(observation),
251                    "question": "Existem observações locais disponíveis para corroborar este sinal?",
252                }
253                event_id = None
254                incident_id = (state.incident_id if isinstance(state, IncidentState)
255                               else f"incident-{observation.event_id}")
256                source_id = observation.source_id
257            else:
258                # Backwards-compatible Etapa 3 demonstration path. The Etapa 4
259                # demonstration always supplies a real ObservationFeed.
260                created_at = 0.0
261                payload = {
262                    "event_type": "suspected_activity",
263                    "position": {"x": 6, "y": 9},
264                    "question": "Existe evidência independente deste evento?",
265                }
266                event_id = "event-example-001"
267                incident_id = "incident-example-001"
268                source_id = "sensor-example"
269
270            request = Message(
271                message_id=f"request-{uuid.uuid4().hex}",
272                message_type=MessageType.VERIFY_REQUEST,
273                created_at=created_at,
274                payload=payload,
275                event_id=event_id,
276                incident_id=incident_id,
277                source_id=source_id,
278                conversation_id=conversation_id,
279            )
280            outgoing = build_spade_message(
281                agent.informant_jid,
282                request,
283                performative=PERFORMATIVE_REQUEST,
284                conversation_id=conversation_id,
285                reply_with=request.message_id,
286            )
287            await self.send(outgoing)
288            logger.info("VERIFY_REQUEST enviada para %s", agent.informant_jid)
289
290            # Reenvio: sem resposta ao fim do prazo curto, o mesmo pedido (mesmo message_id) volta a sair.
291            # Só o iniciador configurado (configure_mvp) traz estes dois valores; sem eles não há reenvio.
292            resends = getattr(agent, "retransmit", 0)
293            resend_after = getattr(agent, "max_receive_timeout", None)
294            if type(resends) is not int or not is_finite_number(resend_after):
295                resends, resend_after = 0, 0.0
296            # Com o prazo do iniciador configurado, a espera é a dele (uma por tentativa), como nos
297            # outros pedidos; RESPONSE_TIMEOUT_SECONDS continua a ser o teto.
298            response_deadline = monotonic() + (min(RESPONSE_TIMEOUT_SECONDS, resend_after * (1 + resends))
299                                               if resend_after else RESPONSE_TIMEOUT_SECONDS)
300            resend_at = monotonic() + resend_after
301            rejected_responses: list[dict[str, object]] = []
302            while True:
303                remaining = response_deadline - monotonic()
304                if remaining <= 0:
305                    break
306                if resends:
307                    remaining = min(remaining, max(resend_at - monotonic(), 0.001))
308                incoming = await self.receive(timeout=remaining)
309                if incoming is None and resends and monotonic() < response_deadline:
310                    resends -= 1
311                    await self.send(outgoing)
312                    resend_at = monotonic() + resend_after
313                    continue
314                if incoming is None or monotonic() >= response_deadline:
315                    break

317                    response = parse_spade_message(incoming)
318                except (TypeError, ValueError) as error:

320                    rejected_responses.append({
321                        "reason": _rejected_response_reason(incoming, request),
322                        "message_id": None,
323                    })
324                    continue
325                if not response_matches_request(
326                    incoming, response, request, agent.informant_jid,
327                    # O runner pode ter avançado este relógio durante o await receive.
328                    # A demonstração antiga sem feed só conhece o instante do pedido.
329                    current_time=(agent.created_at if agent.created_at is not None else request.created_at),
330                ):
331                    logger.warning("VERIFY_RESPONSE rejeitada: correlação ou remetente inválido")
332                    rejected_responses.append({
333                        "reason": "invalid_correlation",
334                        "message_id": response.message_id,
335                    })
336                    continue
337                if observations:
338                    try:
339                        aggregate = aggregate_verification_responses(
340                            observation,
341                            [response],
342                            request.conversation_id,
343                            request.message_id,
344                            agent.max_distance,
345                            agent.max_time_difference,
346                            incompatibility_rules=agent.incompatibility_rules,
347                            compatibility_rules=agent.compatibility_rules,
348                            expected_incident_id=request.incident_id,
349                        )
350                        if isinstance(getattr(agent, "response_policy", None), ResponsePolicy):
351                            evaluation, decision = _evaluate_response_policy(
352                                agent, observation, aggregate, request.incident_id
353                            )
354                        else:
355                            evaluation = evaluate_preliminary_corroboration(
356                                aggregate, agent.eligible_source_ids
357                            )
358                            decision = decide_preliminary_verification(
359                                evaluation, agent.minimum_compatible_sources
360                            )
361                    except (TypeError, ValueError) as error:
362                        logger.warning("Corroboração não avaliada: %s", error)
363                        rejected_responses.append({
364                            "reason": "invalid_response",
365                            "message_id": response.message_id,
366                        })
367                        continue
368                    agent.aggregate_summary = aggregate
369                    agent.evaluation_summary = evaluation
370                    agent.decision = decision
371                    _record_verification_history(
372                        agent, request, observation, aggregate, evaluation, decision
373                    )
374                    _log_corroboration_result(observation, aggregate, evaluation, decision)
375                else:
376                    _record_safe_observe(agent, None, "no_initial_observation", request)
377                agent.response = response
378                logger.info("VERIFY_RESPONSE recebida de %s", incoming.sender)
379                _record_response_wait_diagnostics(
380                    agent, request, rejected_responses, timed_out=False
381                )
382                return
383
384            logger.warning("Timeout à espera de VERIFY_RESPONSE")
385            # Preserve the previous diagnostic reason when no valid reply follows
386            # a rejection, while recording that the global wait did expire.
387            reason = rejected_responses[-1]["reason"] if rejected_responses else "timeout"
388            _record_safe_observe(
389                agent, observation if observations else None, reason, request
390            )
391            _record_response_wait_diagnostics(
392                agent, request, rejected_responses, timed_out=True
393            )
