 48def response_matches_request(
 49    incoming: Any,
 50    response: Message,
 51    request: Message,
 52    expected_sender: str,
 53    *,
 54    current_time: float,
 55) -> bool:
 56    """Validar intenção, correlação e emissão da resposta até ao instante simulado do recetor.
 57
 58    Args:
 59        incoming (Any): Mensagem SPADE recebida; dá o remetente e os metadados de transporte.
 60        response (Message): Corpo de ``incoming`` já interpretado.
 61        request (Message): VERIFY_REQUEST a que a resposta deve corresponder.
 62        expected_sender (str): JID do informador a quem o pedido foi enviado.
 63        current_time (float): Instante simulado do recetor; a resposta não pode ser posterior.
 64
 65    Returns:
 66        bool: True se a resposta for um VERIFY_RESPONSE ``inform`` do informador esperado, da
 67            mesma conversa e incidente, ao mesmo pedido e emitida entre o pedido e ``current_time``.
 68    """
 69    sender = bare_jid(getattr(incoming, "sender", ""))
 70    expected = bare_jid(expected_sender)
 71    return (
 72        response.message_type is MessageType.VERIFY_RESPONSE
 73        and incoming.get_metadata("performative") == PERFORMATIVE_INFORM
 74        and response.conversation_id == request.conversation_id
 75        and response.incident_id == request.incident_id
 76        and is_finite_number(current_time)
 77        and request.created_at <= response.created_at <= current_time
 78        and incoming.get_metadata("conversation-id") == request.conversation_id
 79        and incoming.get_metadata("in-reply-to") == request.message_id
 80        and response.payload.get("request_message_id") == request.message_id
 81        and sender == expected
 82    )

186    def set_response_policy(
187        self, policy: ResponsePolicy, reputation_scores, source_profiles,
188        *, severity: float = 0.5, default_reputation: float = 0.5,
189    ) -> None:
190        """Receber apenas pesos históricos; o sensor não recebe resultados de referência.
191
192        Args:
193            policy (ResponsePolicy): Limiares e mínimos de evidência da decisão de resposta.
194            reputation_scores (Mapping[str, float]): Reputação histórica por fonte, entre 0 e 1.
195            source_profiles (Mapping[str, Mapping[str, str]]): ``modality`` e
196                ``correlation_group`` de cada fonte.
197            severity (float): Gravidade da ocorrência, entre 0 e 1.
198            default_reputation (float): Reputação de uma fonte sem histórico, entre 0 e 1.
199
200        Raises:
201            TypeError: Se ``policy`` não for ``ResponsePolicy``.
202            ValueError: Se a gravidade ou alguma reputação não estiver entre 0 e 1, ou se um
203                perfil de fonte for inválido.
204        """
205        if not isinstance(policy, ResponsePolicy):
206            raise TypeError("policy deve ser ResponsePolicy.")
207        for value in (severity, default_reputation, *reputation_scores.values()):
208            if not is_finite_number(value) or not 0 <= value <= 1:
209                raise ValueError("Gravidade e reputações devem estar entre 0 e 1.")
210        validate_source_profiles(source_profiles)
211        self.response_policy = policy
212        self.reputation_scores = dict(reputation_scores)
213        self.source_profiles = deepcopy(source_profiles)
214        self.severity = severity
215        self.default_reputation = default_reputation

217    class VerificationBehaviour(OneShotBehaviour):
218        """Enviar um VERIFY_REQUEST ao informador e decidir com a resposta."""
219        async def run(self):
220            """Pedir verificação ao informador e guardar a decisão no agente.
221
222            Respostas inválidas ou mal correlacionadas são rejeitadas e a espera
223            continua. Sem resposta válida até ao prazo, fica a decisão de
224            ``_record_safe_observe``.
225
226            Overrides:
227                spade.behaviour.OneShotBehaviour.run
228            """
229            agent: SensorAgent = self.agent  # type: ignore[assignment]
230            conversation_id = f"conversation-{uuid.uuid4().hex}"
231            agent.conversation_id = conversation_id
232
233            # O iniciador de ocorrências já leu o feed: a verificação usa essa leitura e o mesmo incidente.
234            observations = getattr(agent, "incident_observations", None)
235            if not isinstance(observations, tuple) or not observations:
236                observations = _recent_observations(agent.observation_feed)
237            state = getattr(agent, "incident_state", None)
238
239            if observations:
240                observation = observations[0]
241                agent.initial_observation = observation
242                created_at = (
243                    agent.created_at
244                    if isinstance(agent.created_at, (int, float))
245                    else observation.observed_at
246                )
247                payload = {
248                    "observation": model_to_dict(observation),
249                    "hypothesis": build_observation_hypothesis(observation),
250                    "question": "Existem observações locais disponíveis para corroborar este sinal?",
251                }
252                event_id = None
253                incident_id = (state.incident_id if isinstance(state, IncidentState)
254                               else f"incident-{observation.event_id}")
255                source_id = observation.source_id
256            else:
257                # Backwards-compatible Etapa 3 demonstration path. The Etapa 4
258                # demonstration always supplies a real ObservationFeed.
259                created_at = 0.0
260                payload = {
261                    "event_type": "suspected_activity",
262                    "position": {"x": 6, "y": 9},
263                    "question": "Existe evidência independente deste evento?",
264                }
265                event_id = "event-example-001"
266                incident_id = "incident-example-001"
267                source_id = "sensor-example"
268
269            request = Message(
270                message_id=f"request-{uuid.uuid4().hex}",
271                message_type=MessageType.VERIFY_REQUEST,
272                created_at=created_at,
273                payload=payload,
274                event_id=event_id,
275                incident_id=incident_id,
276                source_id=source_id,
277                conversation_id=conversation_id,
278            )
279            outgoing = build_spade_message(
280                agent.informant_jid,
281                request,
282                performative=PERFORMATIVE_REQUEST,
283                conversation_id=conversation_id,
284                reply_with=request.message_id,
285            )
286            await self.send(outgoing)
287            logger.info("VERIFY_REQUEST enviada para %s", agent.informant_jid)
288
289            # Reenvio: sem resposta ao fim do prazo curto, o mesmo pedido (mesmo message_id) volta a sair.
290            # Só o iniciador configurado (configure_mvp) traz estes dois valores; sem eles não há reenvio.
291            resends = getattr(agent, "retransmit", 0)
292            resend_after = getattr(agent, "max_receive_timeout", None)
293            if type(resends) is not int or not is_finite_number(resend_after):
294                resends, resend_after = 0, 0.0
295            # Com o prazo do iniciador configurado, a espera é a dele (uma por tentativa), como nos
296            # outros pedidos; RESPONSE_TIMEOUT_SECONDS continua a ser o teto.
297            response_deadline = monotonic() + (min(RESPONSE_TIMEOUT_SECONDS, resend_after * (1 + resends))
298                                               if resend_after else RESPONSE_TIMEOUT_SECONDS)
299            resend_at = monotonic() + resend_after
300            rejected_responses: list[dict[str, object]] = []
301            while True:
302                remaining = response_deadline - monotonic()
303                if remaining <= 0:
304                    break
305                if resends:
306                    remaining = min(remaining, max(resend_at - monotonic(), 0.001))
307                incoming = await self.receive(timeout=remaining)
308                if incoming is None and resends and monotonic() < response_deadline:
309                    resends -= 1
310                    await self.send(outgoing)
311                    resend_at = monotonic() + resend_after
312                    continue
313                if incoming is None or monotonic() >= response_deadline:
314                    break
315                try:
316                    response = parse_spade_message(incoming)
317                except (TypeError, ValueError) as error:
318                    logger.warning("Resposta rejeitada: %s", error)
319                    rejected_responses.append({
320                        "reason": _rejected_response_reason(incoming, request),
321                        "message_id": None,
322                    })
323                    continue
324                if not response_matches_request(
325                    incoming, response, request, agent.informant_jid,
326                    # O runner pode ter avançado este relógio durante o await receive.
327                    # A demonstração antiga sem feed só conhece o instante do pedido.
328                    current_time=(agent.created_at if agent.created_at is not None else request.created_at),
329                ):
330                    logger.warning("VERIFY_RESPONSE rejeitada: correlação ou remetente inválido")
331                    rejected_responses.append({
332                        "reason": "invalid_correlation",
333                        "message_id": response.message_id,
334                    })
335                    continue
336                if observations:
337                    try:
338                        aggregate = aggregate_verification_responses(
339                            observation,
340                            [response],
341                            request.conversation_id,
342                            request.message_id,
343                            agent.max_distance,
344                            agent.max_time_difference,
345                            incompatibility_rules=agent.incompatibility_rules,
346                            compatibility_rules=agent.compatibility_rules,
347                            expected_incident_id=request.incident_id,
348                        )
349                        if isinstance(getattr(agent, "response_policy", None), ResponsePolicy):
350                            evaluation, decision = _evaluate_response_policy(
351                                agent, observation, aggregate, request.incident_id
352                            )
353                        else:
354                            evaluation = evaluate_preliminary_corroboration(
355                                aggregate, agent.eligible_source_ids
356                            )
357                            decision = decide_preliminary_verification(
358                                evaluation, agent.minimum_compatible_sources
359                            )
360                    except (TypeError, ValueError) as error:
361                        logger.warning("Corroboração não avaliada: %s", error)
362                        rejected_responses.append({
363                            "reason": "invalid_response",
364                            "message_id": response.message_id,
365                        })
366                        continue
367                    agent.aggregate_summary = aggregate
368                    agent.evaluation_summary = evaluation
369                    agent.decision = decision
370                    _record_verification_history(
371                        agent, request, observation, aggregate, evaluation, decision
372                    )
373                    _log_corroboration_result(observation, aggregate, evaluation, decision)
374                else:
375                    _record_safe_observe(agent, None, "no_initial_observation", request)
376                agent.response = response
377                logger.info("VERIFY_RESPONSE recebida de %s", incoming.sender)
378                _record_response_wait_diagnostics(
379                    agent, request, rejected_responses, timed_out=False
380                )
381                return
382
383            logger.warning("Timeout à espera de VERIFY_RESPONSE")
384            # Preserve the previous diagnostic reason when no valid reply follows
385            # a rejection, while recording that the global wait did expire.
386            reason = rejected_responses[-1]["reason"] if rejected_responses else "timeout"
387            _record_safe_observe(
388                agent, observation if observations else None, reason, request
389            )
390            _record_response_wait_diagnostics(
391                agent, request, rejected_responses, timed_out=True
392            )

422def _rejected_response_reason(incoming: Any, request: Message) -> str:
423    """Distinguir um corpo malformado de um corpo válido de outra conversa."""
424    try:
425        candidate = Message.from_json(incoming.body)
426    except (AttributeError, TypeError, ValueError):
427        return "invalid_response"
428
429    transport_conversation_id = incoming.get_metadata("conversation-id")
430    if (
431        candidate.conversation_id != request.conversation_id
432        or transport_conversation_id != request.conversation_id
433    ):
434        return "invalid_correlation"
435    return "invalid_response"

454def _record_safe_observe(agent, observation, reason: str, request: Message) -> None:
455    """Registar falta de resposta; a política pode pedir patrulha pela evidência local."""
456    eligible = getattr(agent, "eligible_source_ids", ())
457    if not isinstance(eligible, (tuple, list, set)):
458        eligible = ()
459    try:
460        if isinstance(observation, Observation) and isinstance(
461            getattr(agent, "response_policy", None), ResponsePolicy
462        ):
463            evaluation, decision = _evaluate_response_policy(
464                agent, observation, {"classification": "no_data"}, request.incident_id
465            )
466            decision["response_status"] = reason
467        else:
468            evaluation = evaluate_preliminary_corroboration({"classification": "no_data"}, eligible)
469            minimum = getattr(agent, "minimum_compatible_sources", None)
470            if not isinstance(minimum, int) or isinstance(minimum, bool) or minimum <= 0:
471                minimum = 1
472            decision = decide_preliminary_verification(evaluation, minimum)
473            decision.update(decision="observe", reason=reason)
474    except (TypeError, ValueError):
475        evaluation = {}
476        decision = {
477            "decision": "observe",
478            "reason": reason,
479            "compatible_observation_ids": (),
480            "compatible_source_ids": (),
481            "preliminary_score": None,
482            "information_coverage": None,
483        }
484    else:
485        agent.aggregate_summary = None
486        agent.evaluation_summary = evaluation
487        _log_corroboration_result(observation, {}, evaluation, decision)
488    agent.decision = decision
489    _record_verification_history(
490        agent, request, observation, {}, evaluation, decision, classification=reason,
491    )

494def _record_verification_history(
495    agent,
496    request: Message,
497    observation,
498    aggregate,
499    evaluation,
500    decision,
501    *,
502    classification: str | None = None,
503) -> None:
504    """Guardar o registo mínimo que explica a decisão de um pedido concluído."""
505    history = getattr(agent, "verification_history", None)
506    if not isinstance(history, dict):
507        history = {}
508        agent.verification_history = history
509
510    conversation_history = history.setdefault(request.conversation_id, {})
511    if request.message_id in conversation_history:
512        return
513
514    initial_observation = None
515    if isinstance(observation, Observation):
516        initial_observation = model_to_dict(observation)
517
518    conversation_history[request.message_id] = {
519        "conversation_id": request.conversation_id,
520        "request_message_id": request.message_id,
521        "initial_observation": initial_observation,
522        "evidence_observation_ids": tuple(
523            aggregate.get("compatible_observation_ids", ())
524        ),
525        "source_ids": tuple(evaluation.get("observed_source_ids", ())),
526        "compatible_source_ids": tuple(
527            decision.get("compatible_source_ids", ())
528        ),
529        "classification": classification or evaluation.get("classification"),
530        "preliminary_score": decision.get("preliminary_score"),
531        "information_coverage": decision.get("information_coverage"),
532        "decision": decision.get("decision", "observe"),
533        "reason": decision.get("reason"),
534        "response_policy": deepcopy(decision) if "suspicion_score" in decision else None,
535    }

538def _evaluate_response_policy(agent, observation, aggregate, incident_id):
539    """Avaliar a corroboração com reputações e devolver ``(avaliação, decisão)``."""
540    evaluation = evaluate_reputation_corroboration(
541        observation, aggregate, agent.eligible_source_ids,
542        agent.reputation_scores, agent.source_profiles,
543        agent.max_distance, agent.max_time_difference,
544        default_reputation=agent.default_reputation,
545    )
546    decision = decide_response(
547        evaluation, agent.response_policy, severity=agent.severity,
548        initial_observation=observation, incident_id=incident_id,
549    )
550    return evaluation, decision

553def _log_corroboration_result(observation, aggregate, evaluation, decision) -> None:
